✓ Scaffolding project in netflix-watch-party...

🎬 Build Your First Msebetsi Solutions Component

Now, let's create the WatchPartyInvite component. This is what users will see when they're invited to join a watch party.

📄 src/components/WatchPartyInvite.jsx
// Msebetsi Watch Party Invitation Component
import React from 'react';
import './WatchPartyInvite.css';

function WatchPartyInvite({ 
  movieTitle, 
  hostName, 
  startTime, 
  participants 
}) {
  return (
    <div className="watch-party-invite">
      <div className="msebetsi-header">
        <h2>🎬 Msebetsi Watch Party</h2>
        <div className="red-dot">LIVE</div>
      </div>
      
      <div className="movie-info">
        <h3>{movieTitle}</h3>
        <p>Hosted by <strong>{hostName}</strong></p>
      </div>
      
      <div className="party-details">
        <div className="detail">
          <span>⏰ Starts:</span>
          <span>{startTime}</span>
        </div>
        <div className="detail">
          <span>👥 Joined:</span>
          <span>{participants} friends</span>
        </div>
      </div>
      
      <button className="Msebetsi-join-btn">
        ▶ Join Watch Party
      </button>
    </div>
  );
}

export default WatchPartyInvite;
