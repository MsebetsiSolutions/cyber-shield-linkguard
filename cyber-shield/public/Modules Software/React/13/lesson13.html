// Netflix Watch Party Context
import React, { createContext, useContext, useReducer, useEffect } from 'react';
import { useWebSocket } from './useWebSocket';

const WatchPartyContext = createContext();

const initialState = {
    participants: [],
    messages: [],
    reactions: [],
    currentStream: 'f1',
    playbackTime: 0,
    isBuffering: false,
    userReactions: new Map(),
    ps5Connected: false
};

function watchPartyReducer(state, action) {
    switch (action.type) {
        case 'ADD_PARTICIPANT':
            return {
                ...state,
                participants: [...state.participants, action.payload]
            };
        case 'ADD_MESSAGE':
            return {
                ...state,
                messages: [...state.messages.slice(-99), action.payload]
            };
        case 'ADD_REACTION':
            return {
                ...state,
                reactions: [...state.reactions, action.payload],
                userReactions: new Map(state.userReactions)
                    .set(action.payload.userId, action.payload.type)
            };
        case 'SYNC_PLAYBACK':
            return {
                ...state,
                playbackTime: action.payload.time,
                isBuffering: false
            };
        case 'PS5_CONNECTED':
            return {
                ...state,
                ps5Connected: true,
                participants: [...state.participants, {
                    id: 'ps5_user',
                    name: 'PlayStation User',
                    avatar: 'PS5'
                }]
            };
        default:
            return state;
    }
}

export function WatchPartyProvider({ children, streamId }) {
    const [state, dispatch] = useReducer(watchPartyReducer, initialState);
    const { connect, send, subscribe } = useWebSocket();

    useEffect(() => {
        // Connect to watch party WebSocket
        connect(`wss://watchparty.netflix.com/${streamId}`);
        
        // Subscribe to real-time events
        subscribe('chat-message', (data) => {
            dispatch({ type: 'ADD_MESSAGE', payload: data });
        });
        
        subscribe('reaction', (data) => {
            dispatch({ type: 'ADD_REACTION', payload: data });
        });
        
        subscribe('playback-sync', (data) => {
            dispatch({ type: 'SYNC_PLAYBACK', payload: data });
        });

        // PlayStation DualSense integration
        if ('getGamepads' in navigator) {
            navigator.getGamepads();
            dispatch({ type: 'PS5_CONNECTED' });
        }
    }, [streamId]);

    const sendMessage = (text) => {
        const message = {
            id: Date.now(),
            text,
            userId: 'current_user',
            timestamp: new Date().toISOString()
        };
        send('chat-message', message);
        dispatch({ type: 'ADD_MESSAGE', payload: message });
    };

    const sendReaction = (type) => {
        const reaction = {
            type,
            userId: 'current_user',
            timestamp: Date.now()
        };
        send('reaction', reaction);
        dispatch({ type: 'ADD_REACTION', payload: reaction });
    };

    const value = {
        state,
        sendMessage,
        sendReaction,
        syncPlayback: (time) => dispatch({ 
            type: 'SYNC_PLAYBACK', 
            payload: { time } 
        })
    };

    return (
        
            {children}
        
    );
}

export const useWatchParty = () => {
    const context = useContext(WatchPartyContext);
    if (!context) {
        throw new Error('useWatchParty must be used within WatchPartyProvider');
    }
    return context;
};
