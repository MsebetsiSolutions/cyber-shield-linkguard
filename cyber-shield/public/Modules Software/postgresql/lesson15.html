-- PostgreSQL Encryption Implementation
CREATE EXTENSION pgcrypto;

-- Column-level encryption
CREATE TABLE sensitive_data (
id SERIAL PRIMARY KEY,
plain_text TEXT,
encrypted_data BYTEA
);

-- Insert encrypted data
INSERT INTO sensitive_data (plain_text, encrypted_data)
VALUES (
'Confidential Information',
pgp_sym_encrypt('Confidential Information', 'encryption_key_123')
);

-- Query encrypted data
SELECT pgp_sym_decrypt(encrypted_data, 'encryption_key_123')
FROM sensitive_data;

-- Comprehensive Security Audit Query
WITH security_checks AS (
-- Check for weak passwords
SELECT 'Weak Password Check' AS check_name,
COUNT(*) AS issues
FROM pg_authid
WHERE passwd IS NULL OR length(passwd) < 12

UNION ALL

-- Check for excessive privileges
SELECT 'Excessive Privileges', COUNT(*)
FROM pg_roles r
JOIN pg_auth_members m ON r.oid = m.roleid
WHERE r.rolsuper = true

UNION ALL

-- Check for unencrypted sensitive data
SELECT 'Unencrypted Sensitive Columns', COUNT(*)
FROM information_schema.columns c
JOIN pg_class cl ON c.table_name = cl.relname
WHERE (c.column_name LIKE '%password%'
OR c.column_name LIKE '%ssn%'
OR c.column_name LIKE '%credit%')
AND c.data_type NOT IN ('bytea', 'text')
)
SELECT * FROM security_checks WHERE issues > 0;
