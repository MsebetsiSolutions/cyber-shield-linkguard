Solution 2: Client Tier System

Business Rules:

  • Basic: Budget < R50,000
  • Professional: R50,000 ≤ Budget < R200,000
  • Enterprise: Budget ≥ R200,000
function assignClientTier(budget) {
  // Validate input is a number
  if (typeof budget !== "number") {
    return "Error: Budget must be a number";
  }
  
  // Check for negative budgets
  if (budget < 0) {
    return "Error: Budget cannot be negative";
  }
  
  // Tier assignment using comparison operators
  if (budget < 50000) {
    return "Basic";
  } else if (budget >= 50000 && budget < 200000) {
    return "Professional";
  } else {
    return "Enterprise";
  }
}

// Test the function

console.log(assignClientTier(75000)); // "Professional"
console.log(assignClientTier(250000)); // "Enterprise"
console.log(assignClientTier("50000")); // Error message

Solution 3: Project Deadline Checker

Business Rules:

  • On track: Days left > 7
  • Warning: 3 ≤ Days left ≤ 7
  • Critical: Days left < 3
  • Overdue: Days left ≤ 0
function checkProjectStatus(daysLeft) {
  // Input validation
  if (typeof daysLeft !== "number") {
    return "Error: Days left must be a number";
  }
  
  // Check status using comparison operators
  if (daysLeft > 7) {
    return "✅ On track";
  } else if (daysLeft >= 3 && daysLeft <= 7) {
    return "⚠️ Warning";
  } else if (daysLeft > 0 && daysLeft < 3) {
    return "🚨 Critical";
  } else if (daysLeft <= 0) {
    return "❌ Overdue";
  }
}

// Test cases
console.log(checkProjectStatus(10)); // "✅ On track"
console.log(checkProjectStatus(5)); // "⚠️ Warning"
console.log(checkProjectStatus(2)); // "🚨 Critical"
console.log(checkProjectStatus(0)); // "❌ Overdue"
console.log(checkProjectStatus(-3)); // "❌ Overdue"

Solution 4: Complete Business Logic System

// Complete Msebetsi Project Approval System
function approveProject(project) {
  // Destructure project data
  const { budget, teamSize, deadlineDays, clientType } = project;
  
  let approval = true;
  let messages = [];
  
  // Budget validation
  if (budget < 10000) {
    approval = false;
    messages.push("❌ Budget below minimum (R10,000)");
  } else if (budget > 500000) {
    approval = false;
    messages.push("❌ Budget exceeds maximum (R500,000)");
  }
  
  // Team size validation
  if (teamSize < 1) {
    approval = false;
    messages.push("❌ Team must have at least 1 member");
  } else if (teamSize > 10) {
    messages.push("⚠️ Large team - monitor closely");
  }
  
  // Deadline validation
  if (deadlineDays <= 0) {
    approval = false;
    messages.push("❌ Project already overdue");
  } else if (deadlineDays < 7) {
    messages.push("⚠️ Tight deadline - prioritize");
  }
  
  // Client type benefits
  if (clientType === "enterprise" && budget >= 100000) {
    messages.push("🎯 Enterprise client - assign senior team");
  }
  
  // Final decision
  if (approval) {
    messages.unshift("✅ PROJECT APPROVED");
  } else {
    messages.unshift("❌ PROJECT REJECTED");
  }
  
  return messages.join("\n");
}

// Test project data
const project1 = {
  budget: 75000,
  teamSize: 3,
  deadlineDays: 30,
  clientType: "professional"
};

const project2 = {
  budget: 5000,
  teamSize: 0,
  deadlineDays: -5,
  clientType: "basic"
};

console.log(approveProject(project1));
console.log("\n---\n");
console.log(approveProject(project2));

🎯 Best Practices Summary

  1. Always use === and !== for predictable comparisons
  2. Validate inputs before making comparisons
  3. Chain comparisons with logical operators (&&, ||)
  4. Use meaningful variable names in business logic
  5. Test edge cases (0, negative numbers, strings vs numbers)
  6. Document business rules as comments in code
