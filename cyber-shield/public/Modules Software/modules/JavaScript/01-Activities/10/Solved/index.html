function evaluateProject() {
  // Current Variables
  const budget = 750000;
  const deadline = 30;
  const teamSize = 5;
  let decision = "";
  
  // CONDITION 1: Budget Check
  if (budget < 50000) {
    decision = "❌ Budget too low";
    activateRedPath();
  }
  
  // CONDITION 2: Team Check
  else if (teamSize < 2) {
    decision = "⚠️ Need more developers";
    activateYellowPath();
  }
  
  // CONDITION 3: Combined Check
  else if (budget >= 500000 && deadline <= 14) {
    decision = "🚨 High pressure project";
    activateYellowPath();
  }
  
  // CONDITION 4: Enterprise Check
  else if (budget > 1000000 || teamSize > 10) {
    decision = "🎯 Enterprise project";
    activateGreenPath();
  }
  
  // DEFAULT: All good
  else {
    decision = "✅ Project approved!";
    activateGreenPath();
  }
  
  return decision;
}
