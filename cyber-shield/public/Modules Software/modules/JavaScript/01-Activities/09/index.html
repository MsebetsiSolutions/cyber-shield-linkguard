function evaluateProject() {
  // Get current values from our traffic control system
  const budget = 750000;
  const deadline = 30;
  const teamSize = 5;
  let decision = "";
  
  // CONDITION 1: Budget check
  if (budget < 50000) {
    decision = "❌ Budget too low";
    setLight("red");
  }
  
  // CONDITION 2: Team capacity check
  else if (teamSize < 2) {
    decision = "⚠️ Need more developers";
    setLight("yellow");
  }
  
  // CONDITION 3: Combined check using logical operators
  else if (budget >= 500000 && deadline <= 14) {
    decision = "🚨 High pressure project";
    setLight("yellow");
  }
  
  // CONDITION 4: Complex condition with OR
  else if (budget > 1000000 || teamSize > 10) {
    decision = "🎯 Enterprise-level project";
    setLight("green");
  }
  
  // DEFAULT: All conditions pass
  else {
    decision = "✅ Project approved!";
    setLight("green");
  }
  
  return decision;
}
