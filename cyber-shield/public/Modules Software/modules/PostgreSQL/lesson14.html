Stored Procedure Fundamentals

Understanding the building blocks of database automation

A stored procedure is a precompiled collection of SQL statements and optional control-flow logic stored in the database. Unlike functions, procedures can manage transactions and don't necessarily return values.

Basic Procedure Structure
-- BASIC PROCEDURE STRUCTURE IN POSTGRESQL
CREATE OR REPLACE PROCEDURE log_security_event(
  p_event_type VARCHAR(50),
  p_severity VARCHAR(20),
  p_source_ip INET,
  p_description TEXT
)
LANGUAGE plpgsql
AS $$
BEGIN
  -- Input validation
  IF p_severity NOT IN ('LOW', 'MEDIUM', 'HIGH', 'CRITICAL') THEN
    RAISE EXCEPTION 'Invalid severity level: %', p_severity;
  END IF;

  -- Core logic
  INSERT INTO security_events (
    event_type,
    severity,
    source_ip,
    description,
    timestamp
  ) VALUES (
    p_event_type,
    p_severity,
    p_source_ip,
    p_description,
    NOW()
  );

  -- Audit logging
  INSERT INTO audit_log (
    action,
    details,
    performed_at
  ) VALUES (
    'SECURITY_EVENT_LOGGED',
    format('Event type: %, Severity: %, IP: %',
      p_event_type, p_severity, p_source_ip),
    NOW()
  );

  RAISE NOTICE 'Security event logged successfully: %', p_event_type;
END;
$$;

MSEBETSI Security Procedure: Threat Response CRITICAL
-- THREAT RESPONSE PROCEDURE: MSEBETSI SECURITY STANDARD
-- Automates response to security incidents with full audit trail

CREATE OR REPLACE PROCEDURE respond_to_threat(
  p_threat_id INTEGER,
  p_response_action VARCHAR(50),
  p_analyst_id INTEGER,
  p_notes TEXT DEFAULT NULL,
  OUT p_response_id INTEGER
)
LANGUAGE plpgsql
SECURITY DEFINER -- Executes with definer's privileges
AS $$
DECLARE
  v_threat_record threats%ROWTYPE;
  v_current_time TIMESTAMPTZ := NOW();
  v_validation_error BOOLEAN := FALSE;
BEGIN
  -- 1. VALIDATION PHASE
  BEGIN
    SELECT * INTO v_threat_record
    FROM threats
    WHERE id = p_threat_id
    FOR UPDATE; -- Lock the record
  
    IF NOT FOUND THEN
      RAISE EXCEPTION 'Threat ID % not found', p_threat_id;
    END IF;
  
    IF v_threat_record.status = 'RESOLVED' THEN
      RAISE EXCEPTION 'Threat % is already resolved', p_threat_id;
    END IF;
  
    IF p_response_action NOT IN (
      'CONTAIN', 'ISOLATE', 'ERADICATE', 'ESCALATE'
    ) THEN
      RAISE EXCEPTION 'Invalid response action: %', p_response_action;
    END IF;
  EXCEPTION
    WHEN OTHERS THEN
      v_validation_error := TRUE;
      RAISE;
  END;

  IF v_validation_error THEN
    RETURN; -- Exit if validation failed
  END IF;

  -- 2. EXECUTION PHASE (within transaction)
  BEGIN
    -- Update threat status
    UPDATE threats
    SET
      status = 'IN_RESPONSE',
      response_action = p_response_action,
      responded_at = v_current_time,
      responding_analyst = p_analyst_id,
      last_updated = v_current_time
    WHERE id = p_threat_id;

    -- Log the response
    INSERT INTO threat_responses (
      threat_id,
      action,
      analyst_id,
      notes,
      created_at
    ) VALUES (
      p_threat_id,
      p_response_action,
      p_analyst_id,
      p_notes,
      v_current_time
    ) RETURNING id INTO p_response_id;

    -- Audit trail
    INSERT INTO security_audit (
      event_type,
      entity_type,
      entity_id,
      details,
      performed_by,
      performed_at
    ) VALUES (
      'THREAT_RESPONSE_INITIATED',
      'THREAT',
      p_threat_id,
      jsonb_build_object(
        'response_action', p_response_action,
        'analyst_id', p_analyst_id,
        'threat_previous_status', v_threat_record.status,
        'response_id', p_response_id
      ),
      p_analyst_id,
      v_current_time
    );

    RAISE NOTICE 'Threat response initiated. Response ID: %', p_response_id;
  EXCEPTION
    WHEN OTHERS THEN
      RAISE EXCEPTION 'Failed to initiate threat response: %', SQLERRM;
  END;
END;
$$;

Transaction Management & Error Handling

Ensuring reliability in security operations

Security procedures must handle failures gracefully while maintaining data consistency. PostgreSQL provides robust transaction control and error handling mechanisms that are crucial for security operations.

Transaction Control in Procedures
-- TRANSACTION MANAGEMENT IN STORED PROCEDURES
-- Critical for maintaining data consistency in security operations

CREATE OR REPLACE PROCEDURE process_security_incident(
  p_incident_data JSONB
)
LANGUAGE plpgsql
AS $$
DECLARE
  v_incident_id INTEGER;
  v_asset_count INTEGER;
  v_error_message TEXT;
  v_operation_start TIMESTAMPTZ := clock_timestamp();
BEGIN
  -- Explicit transaction control (optional but recommended)
  BEGIN
    -- 1. Create incident record
    INSERT INTO security_incidents (
      title,
      description,
      severity,
      reported_at,
      reported_by,
      raw_data
    ) VALUES (
      p_incident_data->>'title',
      p_incident_data->>'description',
      p_incident_data->>'severity',
      NOW(),
      p_incident_data->>'reported_by',
      p_incident_data
    ) RETURNING id INTO v_incident_id;

    -- 2. Link affected assets
    INSERT INTO incident_assets (incident_id, asset_id)
    SELECT v_incident_id, value::INTEGER
    FROM jsonb_array_elements_text(p_incident_data->'affected_assets');
    
    GET DIAGNOSTICS v_asset_count = ROW_COUNT;

    -- 3. Create initial response task
    INSERT INTO incident_tasks (
      incident_id,
      task_type,
      priority,
      assigned_to,
      due_date
    ) VALUES (
      v_incident_id,
      'INITIAL_ASSESSMENT',
      CASE WHEN p_incident_data->>'severity' = 'CRITICAL' THEN 1 ELSE 3 END,
      p_incident_data->>'assigned_analyst',
      NOW() + INTERVAL '2 hours'
    );

    -- 4. Log the operation
    INSERT INTO operation_log (
      operation,
      details,
      start_time,
      end_time,
      status
    ) VALUES (
      'INCIDENT_CREATION',
      jsonb_build_object(
        'incident_id', v_incident_id,
        'asset_count', v_asset_count,
        'severity', p_incident_data->>'severity'
      ),
      v_operation_start,
      clock_timestamp(),
      'SUCCESS'
    );

    RAISE NOTICE 'Incident % created successfully with % affected assets',
      v_incident_id, v_asset_count;
  
    -- Implicit commit happens here if no errors
  EXCEPTION
    WHEN OTHERS THEN
      -- Capture error details
      v_error_message := format('%: %', SQLSTATE, SQLERRM);
      
      -- Log the failure
      INSERT INTO operation_log (
        operation,
        details,
        start_time,
        end_time,
        status,
        error_message
      ) VALUES (
        'INCIDENT_CREATION',
        p_incident_data,
        v_operation_start,
        clock_timestamp(),
        'FAILED',
        v_error_message
      );
  
      -- Re-raise the error for calling application
      RAISE EXCEPTION 'Incident creation failed: %', v_error_message;
  END; -- End of transaction block
END;
$$;

Transaction Best Practices

  • Keep transactions short to reduce locking
  • Use explicit BEGIN/END blocks for complex operations
  • Always include comprehensive error handling
  • Log both successes and failures
  • Consider isolation levels for concurrent operations

Common Pitfalls

  • Long-running transactions blocking other operations
  • Insufficient error handling leading to silent failures
  • Not validating inputs before transaction start
  • Missing audit trails for security operations
  • Forgetting to release advisory locks

Advanced Security Procedure: Dynamic Threat Quarantine
-- ADVANCED PROCEDURE: DYNAMIC THREAT QUARANTINE
-- Uses multiple advanced features for adaptive threat response

CREATE OR REPLACE PROCEDURE quarantine_threat_source(
  p_source_identifier TEXT,
  p_identifier_type VARCHAR(20) DEFAULT 'IP',
  p_quarantine_duration INTERVAL DEFAULT '24 hours',
  p_reason TEXT
)
LANGUAGE plpgsql
SECURITY DEFINER -- Run with elevated privileges
SET search_path = public -- Control search path for security
AS $$
DECLARE
  v_quarantine_id UUID;
  v_affected_count INTEGER := 0;
  v_dynamic_sql TEXT;
  v_table_name TEXT;
  v_column_name TEXT;
  v_quarantine_rec RECORD;
  v_error_context TEXT;
BEGIN
  -- Validate identifier type
  IF p_identifier_type NOT IN ('IP', 'USERNAME', 'EMAIL', 'HOSTNAME') THEN
    RAISE EXCEPTION 'Invalid identifier type: %', p_identifier_type;
  END IF;

  -- Create quarantine record
  INSERT INTO threat_quarantines (
    source_identifier,
    identifier_type,
    quarantined_at,
    expires_at,
    reason,
    status
  ) VALUES (
    p_source_identifier,
    p_identifier_type,
    NOW(),
    NOW() + p_quarantine_duration,
    p_reason,
    'ACTIVE'
  ) RETURNING id, quarantined_at, expires_at
  INTO v_quarantine_rec;

  -- Determine which table/column to update based on identifier type
  SELECT
    CASE p_identifier_type
      WHEN 'IP' THEN 'network_devices'
      WHEN 'USERNAME' THEN 'user_accounts'
      WHEN 'EMAIL' THEN 'email_accounts'
      WHEN 'HOSTNAME' THEN 'servers'
    END,
    CASE p_identifier_type
      WHEN 'IP' THEN 'ip_address'
      WHEN 'USERNAME' THEN 'username'
      WHEN 'EMAIL' THEN 'email'
      WHEN 'HOSTNAME' THEN 'hostname'
    END
  INTO v_table_name, v_column_name;

  -- Build and execute dynamic SQL
  v_dynamic_sql := format(
    'UPDATE %I SET status = $1, quarantine_id = $2, last_updated = NOW() ' ||
    'WHERE %I = $3 AND status != ''QUARANTINED''',
    v_table_name, v_column_name
  );

  -- Execute dynamic SQL with parameters
  EXECUTE v_dynamic_sql
  USING 'QUARANTINED', v_quarantine_rec.id, p_source_identifier;

  GET DIAGNOSTICS v_affected_count = ROW_COUNT;

  -- Log the quarantine action
  INSERT INTO security_audit (
    event_type,
    entity_type,
    entity_id,
    details,
    performed_at
  ) VALUES (
    'THREAT_QUARANTINE',
    'QUARANTINE',
    v_quarantine_rec.id,
    jsonb_build_object(
      'source', p_source_identifier,
      'type', p_identifier_type,
      'affected_count', v_affected_count,
      'duration', p_quarantine_duration,
      'reason', p_reason,
      'expires', v_quarantine_rec.expires_at
    ),
    NOW()
  );

  RAISE NOTICE 'Quarantined %: % affected records. Expires: %',
    p_source_identifier, v_affected_count, v_quarantine_rec.expires_at;

EXCEPTION
  WHEN OTHERS THEN
    -- Get detailed error context
    GET STACKED DIAGNOSTICS v_error_context = PG_EXCEPTION_CONTEXT;
    
    -- Log error with full context
    INSERT INTO procedure_errors (
      procedure_name,
      error_message,
      error_context,
      parameters,
      occurred_at
    ) VALUES (
      'quarantine_threat_source',
      SQLERRM,
      v_error_context,
      jsonb_build_object(
        'source_identifier', p_source_identifier,
        'identifier_type', p_identifier_type,
        'quarantine_duration', p_quarantine_duration
      ),
      NOW()
    );
    
    -- Re-raise with additional context
    RAISE EXCEPTION 'Quarantine failed for % (%): %',
      p_source_identifier, p_identifier_type, SQLERRM;
END;
$$;
