-- MSEBETSI SECURITY ANALYTICS DATABASE SCHEMA
CREATE TABLE security_events (
  event_id SERIAL PRIMARY KEY,
  event_type VARCHAR(50) NOT NULL,
  severity VARCHAR(20) CHECK (severity IN ('LOW', 'MEDIUM', 'HIGH', 'CRITICAL')),
  source_ip INET,
  timestamp TIMESTAMPTZ DEFAULT NOW(),
  description TEXT,
  analyst_id INT REFERENCES security_analysts(id),
  threat_id INT REFERENCES threat_indicators(id)
);

CREATE TABLE threat_indicators (
  id SERIAL PRIMARY KEY,
  indicator_type VARCHAR(50),
  value VARCHAR(500) NOT NULL,
  first_seen TIMESTAMPTZ,
  last_seen TIMESTAMPTZ,
  confidence_score DECIMAL(3,2)
);

CREATE TABLE security_analysts (
  id SERIAL PRIMARY KEY,
  username VARCHAR(50) UNIQUE NOT NULL,
  team VARCHAR(50),
  clearance_level INT DEFAULT 1
);

COUNT() Function Deep Dive

Three variations for different security analytics needs

The COUNT() function returns the number of rows that match a specific condition [citation:7]. Understanding its variations is crucial for accurate security reporting.

COUNT() Variations Comparison
-- 1. COUNT(*): Counts ALL rows, including NULLs and duplicates
-- Security Use: Total events in investigation scope
SELECT COUNT(*) AS total_events
FROM security_events
WHERE timestamp >= '2024-01-01';
-- Result: 10,247 (all rows in timeframe)

-- 2. COUNT(column): Counts non-NULL values in specific column
-- Security Use: Events with complete attribution
SELECT COUNT(source_ip) AS events_with_ip
FROM security_events;
-- Result: 8,912 (excludes NULL source_ip values)

-- 3. COUNT(DISTINCT column): Counts unique non-NULL values
-- Security Use: Unique threat sources or indicators
SELECT COUNT(DISTINCT source_ip) AS unique_attack_sources
FROM security_events
WHERE severity = 'CRITICAL';
-- Result: 142 (unique IPs with critical events)

-- 4. COUNT with FILTER: Advanced conditional counting
SELECT
  COUNT(*) FILTER (WHERE severity = 'CRITICAL') AS critical_count,
  COUNT(*) FILTER (WHERE severity = 'HIGH') AS high_count,
  COUNT(*) FILTER (WHERE event_type LIKE '%SQL_INJECTION%') AS sql_injection_count
FROM security_events
WHERE timestamp >= '2024-01-01';
-- Result: Multiple counts in single query pass

Performance Consideration

Warning: COUNT(*) on large tables requires a full scan in PostgreSQL due to its MVCC implementation. For frequent counting on large security event tables, consider maintaining a counter table or using estimate methods [citation:1][citation:7].

GROUP BY Mastery & HAVING Clause

Categorize and filter aggregated security data

The GROUP BY clause groups rows that have the same values into summary rows [citation:4]. It's particularly useful with aggregate functions to analyze patterns within categories [citation:5].

Security Event Analysis Patterns
-- PATTERN 1: Basic GROUP BY - Events by severity level
SELECT
  severity,
  COUNT(*) AS event_count,
  AVG(EXTRACT(EPOCH FROM (NOW() - timestamp))/3600) AS avg_hours_old
FROM security_events
WHERE timestamp >= '2024-01-01'
GROUP BY severity
ORDER BY event_count DESC;

-- PATTERN 2: GROUP BY multiple columns - Events by type and severity
SELECT
  event_type,
  severity,
  COUNT(*) AS count,
  MIN(timestamp) AS first_occurrence,
  MAX(timestamp) AS last_occurrence
FROM security_events
GROUP BY event_type, severity
ORDER BY count DESC;

-- PATTERN 3: HAVING clause - Filter aggregated results
-- WHERE filters before grouping, HAVING filters after grouping [citation:2]
SELECT
  source_ip,
  COUNT(*) AS attack_count,
  STRING_AGG(DISTINCT event_type, ', ') AS attack_types
FROM security_events
WHERE severity IN ('HIGH', 'CRITICAL')
GROUP BY source_ip
HAVING COUNT(*) >= 5 -- Only IPs with 5+ high/critical events
ORDER BY attack_count DESC;

-- PATTERN 4: GROUP BY with JOIN - Analyst performance metrics
SELECT
  a.username,
  a.team,
  COUNT(e.event_id) AS events_assigned,
  COUNT(e.event_id) FILTER (WHERE e.severity = 'CRITICAL') AS critical_events,
  AVG(EXTRACT(EPOCH FROM (e.resolved_at - e.timestamp))/3600) AS avg_resolution_hours
FROM security_analysts a
LEFT JOIN security_events e ON a.id = e.analyst_id
WHERE e.timestamp >= '2024-01-01'
GROUP BY a.id, a.username, a.team
HAVING COUNT(e.event_id) > 0
ORDER BY events_assigned DESC;
Clause Execution Order Purpose Example
WHERE Before GROUP BY Filter individual rows before aggregation WHERE severity = 'CRITICAL'
GROUP BY After WHERE Group rows for aggregation GROUP BY event_type, source_ip
HAVING After GROUP BY Filter aggregated results HAVING COUNT(*) > 10
ORDER BY Last (usually) Sort final result set ORDER BY event_count DESC

Real-time Dashboard Query Pattern
-- ENTERPRISE SECURITY DASHBOARD QUERY
-- Provides complete situational awareness in single query
WITH event_summary AS (
  SELECT
    severity,
    event_type,
    COUNT(*) AS event_count,
    COUNT(DISTINCT source_ip) AS unique_sources,
    AVG(EXTRACT(EPOCH FROM (NOW() - timestamp))/60) AS avg_minutes_old
  FROM security_events
  WHERE timestamp >= NOW() - INTERVAL '24 HOURS'
  GROUP BY GROUPING SETS (
    (severity, event_type), -- Detail level
    (severity), -- Severity subtotal
    () -- Grand total
  )
)
SELECT
  COALESCE(severity, 'ALL') AS severity_group,
  COALESCE(event_type, 'TOTAL') AS event_type_group,
  event_count,
  unique_sources,
  ROUND(avg_minutes_old, 1) AS avg_age_minutes,
  ROUND(100.0 * event_count / SUM(event_count) OVER (), 2) AS pct_of_total
FROM event_summary
ORDER BY
  CASE WHEN severity = 'CRITICAL' THEN 1
       WHEN severity = 'HIGH' THEN 2
       WHEN severity = 'MEDIUM' THEN 3
       WHEN severity = 'LOW' THEN 4
       ELSE 5 END,
  event_count DESC;
