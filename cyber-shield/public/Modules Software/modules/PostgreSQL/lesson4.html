PART 3: MASTERING CORE OPERATIONS

INSERT Operation

Add new records to your tables:

-- Insert single record
INSERT INTO produce (id, name)
VALUES (1, 'apple');

-- Insert multiple records
INSERT INTO produce (id, name)
VALUES
    (1, 'apple'),
    (2, 'orange'),
    (3, 'banana');

SELECT Operation

Retrieve and filter data:

-- Select all columns
SELECT * FROM produce;

-- Select specific columns
SELECT name, id FROM produce;

-- Select with WHERE clause
SELECT * FROM produce
WHERE id = 1;

UPDATE Operation

Modify existing records:

-- Update single record
UPDATE produce
SET name = 'strawberry'
WHERE id = 1;

-- Update multiple records
UPDATE produce
SET name = 'organic ' || name
WHERE id IN (1, 2);

DELETE Operation

Remove records from tables:

-- Delete specific record
DELETE FROM produce
WHERE id = 2;

-- Delete multiple records
DELETE FROM produce
WHERE name LIKE '%apple%';
CRITICAL: Always use a WHERE clause with DELETE! Omitting WHERE will delete ALL records in the table.
