Create professional blog posts with AI in seconds. Here's the complete system:

Python
import openai
import json
from datetime import datetime
from typing import Dict, List, Optional
import markdown

class BlogGenerator:
"""Generate complete blog posts with AI."""

def __init__(self, api_key: str):
self.client = openai.OpenAI(api_key=api_key)
self.templates = self._load_templates()

def _load_templates(self) -> Dict:
"""Load blog post templates."""
return {
"tutorial": "Write a step-by-step tutorial about {topic}",
"guide": "Create a comprehensive guide about {topic}",
"review": "Write an honest review of {topic}",
"news": "Write a news article about {topic}",
"list": "Create a listicle about {topic}"
}

def generate_outline(self, topic: str, style: str = "tutorial") -> Dict:
"""Generate a blog post outline."""
prompt = f"""Generate a detailed outline for a {style} blog post about "{topic}".
Include:
1. Title
2. Introduction
3. 5-7 main sections with subtopics
4. Conclusion
5. Call to action

Return as JSON with keys: title, introduction, sections, conclusion, cta"""

response = self.client.chat.completions.create(
model="gpt-4",
messages=[{"role": "user", "content": prompt}],
temperature=0.7
)

return json.loads(response.choices[0].message.content)

def generate_full_post(self, outline: Dict) -> str:
"""Generate full blog post from outline."""
sections_text = "\\n".join([
f"{i+1}. {section['title']}: {section.get('description', '')}"
for i, section in enumerate(outline['sections'])
])

prompt = f"""Write a complete blog post based on this outline:

Title: {outline['title']}
Introduction: {outline['introduction']}

Sections:
{sections_text}

Conclusion: {outline['conclusion']}
Call to Action: {outline['cta']}

Write in a professional, engaging tone. Include code examples where relevant.
Use markdown formatting with headings, lists, and code blocks."""

response = self.client.chat.completions.create(
model="gpt-4",
messages=[{"role": "user", "content": prompt}],
temperature=0.8,
max_tokens=2000
)

return response.choices[0].message.content

def save_to_file(self, content: str, filename: str):
"""Save blog post to markdown file."""
with open(filename, 'w', encoding='utf-8') as f:
f.write(content)
print(f"✅ Blog post saved to {filename}")

# Usage example:
blog_gen = BlogGenerator(api_key="your-key-here")
outline = blog_gen.generate_outline("Python AI Development", "tutorial")
post = blog_gen.generate_full_post(outline)
blog_gen.save_to_file(post, "python-ai-tutorial.md")

🎨 Image Generation with DALL-E API

Create stunning images from text descriptions using OpenAI's DALL-E 3:

Python
import openai
import requests
from PIL import Image
from io import BytesIO
import os
from datetime import datetime

class ImageGenerator:
"""Generate and manage images using DALL-E API."""

def __init__(self, api_key: str):
self.client = openai.OpenAI(api_key=api_key)
self.output_dir = "generated_images"
os.makedirs(self.output_dir, exist_ok=True)

def generate_image(self, prompt: str, size: str = "1024x1024",
quality: str = "standard", style: str = "vivid") -> dict:
"""Generate an image from text prompt."""
try:
response = self.client.images.generate(
model="dall-e-3",
prompt=prompt,
size=size,
quality=quality,
style=style,
n=1
)

image_url = response.data[0].url
revised_prompt = response.data[0].revised_prompt

# Download and save the image
filename = self._download_image(image_url, prompt)

return {
"success": True,
"url": image_url,
"revised_prompt": revised_prompt,
"filename": filename,
"size": size,
"timestamp": datetime.now().isoformat()
}

except Exception as e:
return {"success": False, "error": str(e)}

def _download_image(self, url: str, prompt: str) -> str:
"""Download image from URL and save to disk."""
response = requests.get(url)
response.raise_for_status()

# Create filename from prompt
safe_prompt = "".join(c for c in prompt[:50] if c.isalnum() or c in (' ', '-', '_')).rstrip()
timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
filename = f"{safe_prompt}_{timestamp}.png"
filepath = os.path.join(self.output_dir, filename)

# Save image
with open(filepath, 'wb') as f:
f.write(response.content)

return filepath

def generate_variations(self, image_path: str, n: int = 2) -> list:
"""Generate variations of an existing image."""
with open(image_path, "rb") as image_file:
response = self.client.images.create_variation(
image=image_file,
n=n,
size="1024x1024"
)

return [data.url for data in response.data]

def edit_image(self, image_path: str, mask_path: str, prompt: str) -> str:
"""Edit an image using a mask and prompt."""
with open(image_path, "rb") as image_file, open(mask_path, "rb") as mask_file:
response = self.client.images.edit(
image=image_file,
mask=mask_file,
prompt=prompt,
n=1,
size="1024x1024"
)

return response.data[0].url

📱 Automate Social Media Content

Create and schedule social media posts across platforms using AI:

Python
import openai
import json
from datetime import datetime, timedelta
from typing import List, Dict
import csv

class SocialMediaAutomator:
"""Generate and manage social media content."""

def __init__(self, api_key: str):
self.client = openai.OpenAI(api_key=api_key)
self.platform_rules = {
"twitter": {"max_length": 280, "hashtags": 2},
"linkedin": {"max_length": 3000, "hashtags": 3},
"instagram": {"max_length": 2200, "hashtags": 30},
"facebook": {"max_length": 63206, "hashtags": 0}
}

def generate_campaign(self, topic: str, platform: str,
num_posts: int = 5) -> List[Dict]:
"""Generate a social media campaign."""
rules = self.platform_rules[platform]

prompt = f"""Generate {num_posts} social media posts for {platform} about "{topic}".
Platform rules:
- Maximum {rules['max_length']} characters per post
- Use {rules['hashtags']} relevant hashtags per post
- Make each post engaging and platform-appropriate
- Include emojis where appropriate
- Vary the post types (questions, facts, tips, etc.)

Return as JSON list with keys: content, hashtags, post_type"""

response = self.client.chat.completions.create(
model="gpt-4",
messages=[{"role": "user", "content": prompt}],
temperature=0.8
)

posts = json.loads(response.choices[0].message.content)

# Add scheduling dates
start_date = datetime.now()
for i, post in enumerate(posts):
post_date = start_date + timedelta(days=i)
post["scheduled_date"] = post_date.strftime("%Y-%m-%d")
post["platform"] = platform

return posts

def export_to_csv(self, posts: List[Dict], filename: str):
"""Export posts to CSV for scheduling tools."""
fieldnames = ["platform", "content", "hashtags", "post_type", "scheduled_date"]

with open(filename, 'w', newline='', encoding='utf-8') as f:
writer = csv.DictWriter(f, fieldnames=fieldnames)
writer.writeheader()
writer.writerows(posts)

print(f"✅ Exported {len(posts)} posts to {filename}")

def create_content_calendar(self, topics: List[str], platforms: List[str],
posts_per_topic: int = 3) -> Dict:
"""Create a complete content calendar."""
calendar = {}

for topic in topics:
calendar[topic] = {}
for platform in platforms:
posts = self.generate_campaign(topic, platform, posts_per_topic)
calendar[topic][platform] = posts

return calendar

# Usage example:
automator = SocialMediaAutomator(api_key="your-key-here")
posts = automator.generate_campaign("Python AI Development", "twitter", 5)
automator.export_to_csv(posts, "social_media_schedule.csv")

import json
import csv
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List

class ContentManager:
"""Manage AI-generated content in files."""

def __init__(self, data_dir: str = "ai_content"):
self.data_dir = Path(data_dir)
self.data_dir.mkdir(exist_ok=True)

# File paths
self.blog_file = self.data_dir / "blog_posts.json"
self.images_file = self.data_dir / "generated_images.json"
self.social_file = self.data_dir / "social_schedule.csv"

def save_blog_post(self, post_data: Dict[str, Any]) -> str:
"""Save blog post to JSON file."""
# Load existing posts
posts = self._load_json(self.blog_file, default=[])

# Add metadata
post_data["id"] = f"bp_{len(posts) + 1:03d}"
post_data["created_at"] = datetime.now().isoformat()
post_data["updated_at"] = datetime.now().isoformat()

# Add to list and save
posts.append(post_data)
self._save_json(self.blog_file, posts)

return post_data["id"]

def save_image_record(self, image_data: Dict[str, Any]) -> str:
"""Save image generation record."""
images = self._load_json(self.images_file, default=[])

image_data["id"] = f"img_{len(images) + 1:03d}"
image_data["created_at"] = datetime.now().isoformat()

images.append(image_data)
self._save_json(self.images_file, images)

return image_data["id"]

def export_social_schedule(self, posts: List[Dict[str, Any]]) -> str:
"""Export social media posts to CSV."""
fieldnames = ["id", "platform", "content", "hashtags",
"post_type", "scheduled_date", "status"]

# Add IDs to posts
for i, post in enumerate(posts):
post["id"] = f"sm_{i + 1:03d}"
post["status"] = "scheduled"

with open(self.social_file, 'w', newline='', encoding='utf-8') as f:
writer = csv.DictWriter(f, fieldnames=fieldnames)
writer.writeheader()
writer.writerows(posts)

return str(self.social_file)

def _load_json(self, filepath: Path, default: Any = None) -> Any:
"""Load JSON file with error handling."""
if not filepath.exists():
return default if default is not None else {}

try:
with open(filepath, 'r', encoding='utf-8') as f:
return json.load(f)
except json.JSONDecodeError:
return default if default is not None else {}

def _save_json(self, filepath: Path, data: Any):
"""Save data to JSON file."""
with open(filepath, 'w', encoding='utf-8') as f:
json.dump(data, f, indent=2, ensure_ascii=False)

import os
from dotenv import load_dotenv
from typing import Optional
from pathlib import Path

class ConfigManager:
"""Manage application configuration and environment variables."""

def __init__(self, env_file: str = ".env"):
self.env_file = Path(env_file)
self.config = {}
self._load_config()

def _load_config(self):
"""Load configuration from environment variables."""
# Load .env file if it exists
if self.env_file.exists():
load_dotenv(self.env_file)

# Load all relevant environment variables
self.config = {
"openai_api_key": self._get_env("OPENAI_API_KEY", required=True),
"news_api_key": self._get_env("NEWS_API_KEY"),
"default_model": self._get_env("DEFAULT_MODEL", "gpt-4"),
"max_tokens": int(self._get_env("MAX_TOKENS", "2000")),
"temperature": float(self._get_env("TEMPERATURE", "0.7")),
"log_level": self._get_env("LOG_LEVEL", "INFO"),
"data_dir": self._get_env("DATA_DIR", "ai_content")
}

def _get_env(self, key: str, default: Optional[str] = None,
required: bool = False) -> str:
"""Get environment variable with validation."""
value = os.getenv(key, default)

if required and not value:
raise ValueError(f"Environment variable {key} is required but not set")

return value

def get(self, key: str, default: Optional[str] = None) -> str:
"""Get configuration value."""
return self.config.get(key, default)

def create_env_template(self):
"""Create a .env.template file for documentation."""
template = """# AI Content Generator Configuration
# Copy this to .env and fill in your values

# OpenAI API Key (required)

OPENAI_API_KEY=your_openai_api_key_here

# News API Key (optional)
# Get from: https://newsapi.org
NEWS_API_KEY=your_news_api_key_here

# AI Model Settings
DEFAULT_MODEL=gpt-4
MAX_TOKENS=2000
TEMPERATURE=0.7

# Application Settings
LOG_LEVEL=INFO
DATA_DIR=ai_content

# Never commit .env to version control!
# Add .env to .gitignore
"""

from typing import Dict, Any
import json
from datetime import datetime

class ContentPipeline:
"""Complete AI content generation pipeline."""

def __init__(self, blog_generator, image_generator,
social_automator, content_manager):
self.blog_generator = blog_generator
self.image_generator = image_generator
self.social_automator = social_automator
self.content_manager = content_manager

def execute(self, topic: str, platforms: list = None) -> Dict[str, Any]:
"""Execute complete content pipeline."""
print(f"🚀 Starting content pipeline for: {topic}")

# 1. Generate blog content
print("📝 Generating blog content...")
outline = self.blog_generator.generate_outline(topic)
blog_post = self.blog_generator.generate_full_post(outline)

# 2. Generate featured image
print("🎨 Creating featured image...")
image_prompt = f"Featured image for blog post about {topic}, professional, digital art"
image_result = self.image_generator.generate_image(image_prompt)

# 3. Generate social media posts
print("📱 Creating social media content...")
platforms = platforms or ["twitter", "linkedin", "instagram"]
social_posts = []
for platform in platforms:
posts = self.social_automator.generate_campaign(topic, platform, 3)
social_posts.extend(posts)

# 4. Save everything
print("💾 Saving all content...")
blog_id = self.content_manager.save_blog_post({
"title": outline["title"],
"content": blog_post,
"topic": topic,
"outline": outline
})

image_id = self.content_manager.save_image_record(image_result)

csv_path = self.content_manager.export_social_schedule(social_posts)

# 5. Return results
return {
"blog_post": {
"id": blog_id,
"title": outline["title"],
"word_count": len(blog_post.split())
},
"image": {
"id": image_id,
"url": image_result.get("url"),
"prompt": image_prompt
},
"social_media": {
"total_posts": len(social_posts),
"platforms": platforms,
"schedule_file": csv_path
},
"timestamp": datetime.now().isoformat()
}
