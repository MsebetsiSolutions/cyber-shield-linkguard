# Using dict() constructor
coding_agent = dict(
name="CodeBot",
role="developer",

)

agent_scores = {f"agent_{i}": i * 10 for i in range(1, 6)}
# → {'agent_1': 10, 'agent_2': 20, ..., 'agent_5': 50}

# research_agent.clear() # Uncomment to clear

🔄 Looping Through Dictionaries - AI Agent Iteration Patterns

Master different ways to iterate through dictionaries for data processing, filtering, and transformation in AI systems.

Python - Dictionary Looping Mastery
# ==========================================
# DICTIONARY LOOPING PATTERNS FOR AI AGENTS
# ==========================================

# Sample AI agent data
agents = {
"research_bot": {
"name": "Research Assistant",
"status": "active",
"tasks_completed": 42,
"skills": ["analysis", "summarization", "research"]
},
"code_bot": {
"name": "Code Generator",
"status": "busy",
"tasks_completed": 156,
"skills": ["python", "javascript", "sql"]
},
"debug_bot": {
"name": "Debug Assistant",
"status": "idle",
"tasks_completed": 89,
"skills": ["debugging", "testing", "optimization"]
}
}

# 1. BASIC LOOPING PATTERNS
print("=== BASIC LOOPING PATTERNS ===")

# Loop through keys only
print("1. Agent IDs (keys):")
for agent_id in agents:
print(f" • {agent_id}")

# Loop through keys explicitly
print("\n2. Using .keys() method:")
for agent_id in agents.keys():
print(f" • {agent_id}: {agents[agent_id]['name']}")

# Loop through values only
print("\n3. Agent data (values):")
for agent_data in agents.values():
print(f" • {agent_data['name']} - {agent_data['status']}")

# Loop through key-value pairs
print("\n4. Key-value pairs (.items()):")
for agent_id, agent_data in agents.items():
print(f" • {agent_id}: {agent_data['name']} ({agent_data['status']})")

# 2. FILTERING WHILE LOOPING
print("\n=== FILTERING PATTERNS ===")

# Filter active agents
print("1. Active agents only:")
for agent_id, agent_data in agents.items():
if agent_data["status"] == "active":
print(f" • {agent_data['name']} is active")

# Filter by task count
print("\n2. High-performing agents (100+ tasks):")
for agent_id, agent_data in agents.items():
if agent_data["tasks_completed"] > 100:
print(f" • {agent_data['name']}: {agent_data['tasks_completed']} tasks")

# Filter by skills
print("\n3. Agents with Python skill:")
for agent_id, agent_data in agents.items():
if "python" in agent_data["skills"]:
print(f" • {agent_data['name']} knows Python")

# 3. TRANSFORMATION PATTERNS
print("\n=== TRANSFORMATION PATTERNS ===")

# Create new dictionary with transformed values
agent_names = {agent_id: agent_data["name"].upper()
for agent_id, agent_data in agents.items()}
print("1. Uppercase agent names:", agent_names)

# Filter and transform in one step
active_agents = {agent_id: agent_data["name"]
for agent_id, agent_data in agents.items()
if agent_data["status"] == "active"}
print("2. Active agent names:", active_agents)

# Calculate statistics
total_tasks = sum(agent_data["tasks_completed"]
for agent_data in agents.values())
print(f"3. Total tasks completed: {total_tasks}")

# 4. NESTED LOOPING PATTERNS
print("\n=== NESTED LOOPING ===")

# Loop through agents and their skills
print("1. All agent skills:")
for agent_id, agent_data in agents.items():
print(f"\n {agent_data['name']}:")
for skill in agent_data["skills"]:
print(f" • {skill}")

# 5. REAL-WORLD AUTOMATION EXAMPLES
print("\n=== AUTOMATION EXAMPLES ===")

# Example 1: Agent status report
print("1. Agent Status Report:")
for agent_id, agent_data in agents.items():
status_icon = "✅" if agent_data["status"] == "active" else "⚠️"
print(f"{status_icon} {agent_data['name']}: {agent_data['status']}")

# Example 2: Performance dashboard
print("\n2. Performance Dashboard:")
for agent_id, agent_data in agents.items():
performance = agent_data["tasks_completed"] / 200 * 100
print(f"{agent_data['name']}: {'█' * int(performance/10)}{'░' * (10 - int(performance/10))} {performance:.1f}%")

# Example 3: Skill matrix
print("\n3. Skill Matrix:")
all_skills = set()
for agent_data in agents.values():
all_skills.update(agent_data["skills"])

for skill in sorted(all_skills):
skilled_agents = [agent_data["name"]
for agent_data in agents.values()
if skill in agent_data["skills"]]
print(f"{skill}: {', '.join(skilled_agents)}")

# 6. ADVANCED PATTERNS
print("\n=== ADVANCED PATTERNS ===")

# Using enumerate with dictionary items
print("1. Enumerated agent list:")
for i, (agent_id, agent_data) in enumerate(agents.items(), 1):
print(f"{i}. {agent_data['name']} ({agent_id})")

# Sorting while looping
print("\n2. Agents sorted by task count:")
for agent_id, agent_data in sorted(agents.items(),
key=lambda x: x[1]["tasks_completed"],
reverse=True):
print(f" {agent_data['name']}: {agent_data['tasks_completed']} tasks")

# Conditional dictionary building
agent_categories = {}
for agent_id, agent_data in agents.items():
category = "high_performer" if agent_data["tasks_completed"] > 100 else "standard"
agent_categories.setdefault(category, []).append(agent_data["name"])

print("\n3. Agent categories:", agent_categories)

print("\n✅ Dictionary looping mastery complete!")

def get_nested_value(data, keys, default=None):
"""Safely get value from nested dictionary."""

for key in keys:

current = current[key]

return default
return current

Create beautiful, readable output with f-strings, format() method, and template strings for AI agent reports and interfaces.

Python - String Formatting Mastery
# ==========================================
# ADVANCED STRING FORMATTING FOR AI AGENTS
# ==========================================

from datetime import datetime
from string import Template

# Sample AI agent data
agent = {
"id": "agent_001",
"name": "Research Assistant Pro",
"status": "active",
"performance": {
"accuracy": 0.923456,
"speed": 4.2367,
"tasks_completed": 156,
"success_rate": 0.95
},
"settings": {
"model": "gpt-4",
"temperature": 0.7,
"max_tokens": 4000
},
"last_active": datetime.now()
}

# 1. F-STRINGS (MODERN PYTHON)
print("=== F-STRINGS (MODERN PYTHON) ===")

# Basic f-string
print(f"1. Agent: {agent['name']}")
print(f" Status: {agent['status'].upper()}")

# Expressions in f-strings
tasks = agent["performance"]["tasks_completed"]
accuracy = agent["performance"]["accuracy"]
print(f"2. Performance: {tasks} tasks with {accuracy:.1%} accuracy")

# Number formatting
print(f"3. Formatted numbers:")
print(f" • Accuracy: {accuracy:.2%}") # 92.35%
print(f" • Speed: {agent['performance']['speed']:.1f}s") # 4.2s
print(f" • Temperature: {agent['settings']['temperature']:.2f}") # 0.70

# Date formatting
print(f"4. Last active: {agent['last_active']:%Y-%m-%d %H:%M}")
print(f" Date only: {agent['last_active']:%A, %B %d, %Y}")

# Alignment and padding
print("5. Aligned output:")
print(f" Name: {agent['name']:>30}") # Right aligned in 30 chars
print(f" ID: {agent['id']:<30}") # Left aligned in 30 chars
print(f" Status: {agent['status']:^30}") # Center aligned in 30 chars

# 2. .FORMAT() METHOD (LEGACY STYLE)
print("\n=== .FORMAT() METHOD ===")

# Positional arguments
print("1. Agent {} is {}".format(agent["name"], agent["status"]))

# Named arguments
print("2. Agent {name} has {tasks} tasks".format(
name=agent["name"],
tasks=agent["performance"]["tasks_completed"]
))

# Dictionary unpacking with .format()
print("3. Performance: {accuracy:.1%}".format(
accuracy=agent["performance"]["accuracy"]
))

# 3. TEMPLATE STRINGS
print("\n=== TEMPLATE STRINGS ===")

# Create template
report_template = Template("""
AGENT REPORT
============
Name: $name
Status: $status
Tasks Completed: $tasks
Accuracy: ${accuracy}%
""")

# Fill template with data
report = report_template.substitute(
name=agent["name"],
status=agent["status"],
tasks=agent["performance"]["tasks_completed"],
accuracy=agent["performance"]["accuracy"] * 100
)
print(report)

# 4. ADVANCED FORMATTING PATTERNS
print("\n=== ADVANCED PATTERNS ===")

# Multi-line f-strings
report = f'''

{'=' * 30}

⚙️ Settings:

• Temperature: {agent['settings']['temperature']}
• Max Tokens: {agent['settings']['max_tokens']:,}

from datetime import datetime
from typing import Dict, List, Any, Optional
import json
import random

class AIAutomationSystem:
"""Complete AI agent automation system."""

def __init__(self, system_name: str):
# System configuration using dictionaries
self.config = {
"system_name": system_name,
"version": "1.0.0",
"created": datetime.now().isoformat(),
"settings": {
"auto_start": True,
"max_agents": 10,
"log_level": "info",
"backup_interval": 3600
}
}

# Agents dictionary - key: agent_id, value: agent_data
self.agents: Dict[str, Dict] = {}

# Task queue as list of dictionaries
self.task_queue: List[Dict] = []

# Results storage
self.results: Dict[str, List] = {
"completed": [],
"failed": [],
"in_progress": []
}

self.initialize_default_agents()

def initialize_default_agents(self):
"""Initialize default AI agents."""
default_agents = [
{
"id": "research_001",
"name": "Research Assistant",
"type": "research",
"status": "idle",
"capabilities": ["web_search", "analysis", "summarization"],
"performance": {"accuracy": 0.0, "tasks": 0}
},
{
"id": "code_001",
"name": "Code Generator",
"type": "coding",
"status": "idle",
"capabilities": ["python", "debugging", "testing"],
"performance": {"accuracy": 0.0, "tasks": 0}
}
]

for agent in default_agents:
self.agents[agent["id"]] = agent

def add_task(self, task_type: str, description: str, priority: int = 1) -> str:
"""Add a new task to the queue."""
task_id = f"task_{len(self.task_queue) + 1:03d}"
task = {
"id": task_id,
"type": task_type,
"description": description,
"priority": priority,
"status": "pending",
"created": datetime.now().isoformat(),
"assigned_agent": None
}

self.task_queue.append(task)
self.task_queue.sort(key=lambda x: x["priority"], reverse=True)
return task_id

def assign_task(self, task_id: str) -> Optional[str]:
"""Assign a task to an available agent."""
# Find task
task = next((t for t in self.task_queue if t["id"] == task_id), None)
if not task or task["status"] != "pending":
return None

# Find suitable agent
agent_type_map = {
"research": "research",
"coding": "coding",
"debug": "coding",
"documentation": "research"
}

required_type = agent_type_map.get(task["type"], "research")

# Find idle agent of required type
for agent_id, agent in self.agents.items():
if agent["type"] == required_type and agent["status"] == "idle":
# Assign task
task["status"] = "assigned"
task["assigned_agent"] = agent_id
task["assigned_at"] = datetime.now().isoformat()

# Update agent
agent["status"] = "busy"
agent["current_task"] = task_id

self.results["in_progress"].append(task_id)
return agent_id

return None

def complete_task(self, task_id: str, result: Dict[str, Any]) -> bool:
"""Mark a task as completed."""
# Find task
task_index = next((i for i, t in enumerate(self.task_queue)
if t["id"] == task_id), None)

if task_index is None:
return False

task = self.task_queue[task_index]
agent_id = task["assigned_agent"]

if not agent_id or agent_id not in self.agents:
return False

# Update task
task["status"] = "completed"
task["completed_at"] = datetime.now().isoformat()
task["result"] = result

# Update agent
agent = self.agents[agent_id]
agent["status"] = "idle"
agent["performance"]["tasks"] += 1
agent["performance"]["accuracy"] = (
agent["performance"]["accuracy"] * (agent["performance"]["tasks"] - 1) +
result.get("score", 0.5)
) / agent["performance"]["tasks"]
agent.pop("current_task", None)

# Move to completed
if task_id in self.results["in_progress"]:
self.results["in_progress"].remove(task_id)
self.results["completed"].append(task_id)

return True

def generate_report(self) -> str:
"""Generate a formatted system report."""
# Calculate statistics
total_tasks = len(self.task_queue)
completed_tasks = len([t for t in self.task_queue if t["status"] == "completed"])
pending_tasks = len([t for t in self.task_queue if t["status"] == "pending"])

active_agents = len([a for a in self.agents.values() if a["status"] == "busy"])
idle_agents = len([a for a in self.agents.values() if a["status"] == "idle"])

# Format report using f-strings
report = f'''
{'=' * 60}
🤖 AI AUTOMATION SYSTEM REPORT
{self.config['system_name']} - Version {self.config['version']}
{'=' * 60}

📊 SYSTEM OVERVIEW
{'-' * 40}
• Created: {datetime.fromisoformat(self.config['created']).strftime('%Y-%m-%d %H:%M')}
• Total Agents: {len(self.agents)}
• Active Agents: {active_agents}
• Idle Agents: {idle_agents}

📋 TASK MANAGEMENT
{'-' * 40}
• Total Tasks: {total_tasks}
• Completed: {completed_tasks}
• Pending: {pending_tasks}
• Completion Rate: {completed_tasks/total_tasks:.1% if total_tasks > 0 else '0%'}

👥 AGENT PERFORMANCE
{'-' * 40}
'''

# Add agent details
for agent_id, agent in self.agents.items():
status_icon = "🟢" if agent["status"] == "idle" else "🟡"
accuracy = agent["performance"]["accuracy"]
accuracy_bar = "█" * int(accuracy * 10) + "░" * (10 - int(accuracy * 10))

report += f'''
{status_icon} {agent['name']} ({agent_id})
Type: {agent['type'].title()}
Status: {agent['status'].upper()}
Tasks: {agent['performance']['tasks']}
Accuracy: {accuracy_bar} {accuracy:.1%}
'''

report += f'''
{'=' * 60}
Report generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
{'=' * 60}
'''

return report

def to_dict(self) -> Dict[str, Any]:
"""Convert system to dictionary for serialization."""
return {
"config": self.config,
"agents": self.agents,
"task_queue": self.task_queue,
"results": self.results,
"summary": {
"total_agents": len(self.agents),
"total_tasks": len(self.task_queue),
"completed_tasks": len(self.results["completed"])
}
}

def save_to_file(self, filename: str):
"""Save system state to JSON file."""
with open(filename, 'w') as f:
json.dump(self.to_dict(), f, indent=2, default=str)

def load_from_file(self, filename: str):
"""Load system state from JSON file."""
with open(filename, 'r') as f:
data = json.load(f)
self.config = data["config"]
self.agents = data["agents"]
self.task_queue = data["task_queue"]
self.results = data["results"]
