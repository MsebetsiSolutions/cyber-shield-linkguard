📐 Vector Mathematics for AI

Vectors are the fundamental building blocks of modern AI. They represent data points in high-dimensional spaces where similarity search and clustering happen.

Python - Complete Vector Operations
# ==========================================
# VECTOR MATHEMATICS FOR AI/ML
# ==========================================

import numpy as np
from math import sqrt, acos, pi

class VectorOperations:
"""Complete vector mathematics for AI applications."""

@staticmethod
def dot_product(vec1, vec2):
"""Calculate dot product of two vectors."""
return sum(a * b for a, b in zip(vec1, vec2))

@staticmethod
def magnitude(vector):
"""Calculate magnitude (length) of a vector."""
return sqrt(sum(x ** 2 for x in vector))

@staticmethod
def cosine_similarity(vec1, vec2):
"""Calculate cosine similarity between vectors."""
dot = VectorOperations.dot_product(vec1, vec2)
mag1 = VectorOperations.magnitude(vec1)
mag2 = VectorOperations.magnitude(vec2)

if mag1 == 0 or mag2 == 0:
return 0

return dot / (mag1 * mag2)

@staticmethod
def euclidean_distance(vec1, vec2):
"""Calculate Euclidean distance between vectors."""
return sqrt(sum((a - b) ** 2 for a, b in zip(vec1, vec2)))

@staticmethod
def normalize(vector):
"""Normalize a vector to unit length."""
mag = VectorOperations.magnitude(vector)
if mag == 0:
return [0] * len(vector)
return [x / mag for x in vector]

@staticmethod
def angle_between(vec1, vec2):
"""Calculate angle between two vectors in radians."""
cos_sim = VectorOperations.cosine_similarity(vec1, vec2)
# Clamp to [-1, 1] to avoid floating point errors
cos_sim = max(-1, min(1, cos_sim))
return acos(cos_sim)

🧠 AI Embeddings: Turning Data into Vectors

Embeddings transform words, images, and concepts into numerical vectors that capture semantic meaning and relationships.

Python - Creating and Using Embeddings
# ==========================================
# AI EMBEDDINGS: TEXT, IMAGE & MULTIMODAL
# ==========================================

import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

class AIEmbeddings:
"""Create and manipulate AI embeddings."""

@staticmethod
def create_word2vec_style_embeddings(words, embedding_dim=100):
"""Create Word2Vec style embeddings."""
np.random.seed(42)
embeddings = {}

for word in words:
# Generate random embedding (in real Word2Vec, these are learned)
embedding = np.random.randn(embedding_dim)
# Normalize
embedding = embedding / np.linalg.norm(embedding)
embeddings[word] = embedding

return embeddings

@staticmethod
def create_semantic_relationships(embeddings):
"""Create semantic relationships between words."""
# king - man + woman = queen analogy
if all(word in embeddings for word in ["king", "man", "woman"]):
queen_vector = (
embeddings["king"]
- embeddings["man"]
+ embeddings["woman"]
)
embeddings["queen_predicted"] = queen_vector / np.linalg.norm(queen_vector)

return embeddings

@staticmethod
def create_tfidf_embeddings(documents):
"""Create TF-IDF embeddings for documents."""
vectorizer = TfidfVectorizer(max_features=100)
tfidf_matrix = vectorizer.fit_transform(documents)
return tfidf_matrix.toarray(), vectorizer.get_feature_names_out()

@staticmethod
def reduce_dimensions(embeddings, n_components=2):
"""Reduce embedding dimensions for visualization."""
pca = PCA(n_components=n_components)
reduced = pca.fit_transform(embeddings)
return reduced, pca.explained_variance_ratio_

@staticmethod
def find_similar_words(query_word, embeddings, top_n=5):
"""Find most similar words to query word."""
if query_word not in embeddings:
return []

query_vec = embeddings[query_word]
similarities = []

for word, vec in embeddings.items():
if word != query_word:
sim = np.dot(query_vec, vec) / (
np.linalg.norm(query_vec) * np.linalg.norm(vec)
)
similarities.append((word, sim))

# Sort by similarity
similarities.sort(key=lambda x: x[1], reverse=True)
return similarities[:top_n]

class MultimodalEmbedding:
"""Combine text and image embeddings."""

def __init__(self, text_dim=768, image_dim=512, multimodal_dim=1024):
self.text_dim = text_dim
self.image_dim = image_dim
self.multimodal_dim = multimodal_dim

def create_multimodal_embedding(self, text_embedding, image_embedding):
"""Combine text and image embeddings."""
# In practice, this would be a learned transformation
combined = np.concatenate([text_embedding, image_embedding])
# Ensure correct dimension
if len(combined) > self.multimodal_dim:
combined = combined[:self.multimodal_dim]
elif len(combined) < self.multimodal_dim:
# Pad with zeros
combined = np.pad(combined, (0, self.multimodal_dim - len(combined)))

return combined / np.linalg.norm(combined)

🗄️ Vector Databases for AI Applications

Vector databases efficiently store and query high-dimensional embeddings, enabling semantic search, recommendations, and AI memory systems.

Python - Vector Database Implementation
# ==========================================
# VECTOR DATABASE IMPLEMENTATION
# ==========================================

import numpy as np
from typing import List, Tuple, Dict, Any
import json
from datetime import datetime

class VectorDatabase:
"""Simple vector database for AI applications."""

def __init__(self, dimension: int):
self.dimension = dimension
self.vectors = [] # List of vectors
self.metadata = [] # List of metadata dictionaries
self.id_counter = 0

def add_vector(self, vector: List[float], metadata: Dict[str, Any] = None) -> int:
"""Add a vector to the database."""
if len(vector) != self.dimension:
raise ValueError(f"Vector must have dimension {self.dimension}")

vector_id = self.id_counter
self.vectors.append(np.array(vector))

if metadata is None:
metadata = {}

metadata.update({
"id": vector_id,
"added_at": datetime.now().isoformat()
})
self.metadata.append(metadata)

self.id_counter += 1
return vector_id

def similarity_search(self, query_vector: List[float], top_k: int = 5) -> List[Tuple[int, float, Dict]]:
"""Find most similar vectors to query."""
if len(query_vector) != self.dimension:
raise ValueError(f"Query vector must have dimension {self.dimension}")

query_np = np.array(query_vector)
similarities = []

for i, vector in enumerate(self.vectors):
# Calculate cosine similarity
similarity = np.dot(query_np, vector) / (
np.linalg.norm(query_np) * np.linalg.norm(vector)
)
similarities.append((i, similarity))

# Sort by similarity (highest first)
similarities.sort(key=lambda x: x[1], reverse=True)

# Return top_k results with metadata
results = []
for i, similarity in similarities[:top_k]:
results.append((i, similarity, self.metadata[i]))

return results

def euclidean_search(self, query_vector: List[float], top_k: int = 5) -> List[Tuple[int, float, Dict]]:
"""Find nearest vectors using Euclidean distance."""
query_np = np.array(query_vector)
distances = []

for i, vector in enumerate(self.vectors):
distance = np.linalg.norm(query_np - vector)
distances.append((i, distance))

# Sort by distance (lowest first)
distances.sort(key=lambda x: x[1])

results = []
for i, distance in distances[:top_k]:
results.append((i, distance, self.metadata[i]))

return results

def get_by_id(self, vector_id: int) -> Tuple[np.ndarray, Dict]:
"""Get vector and metadata by ID."""
for i, metadata in enumerate(self.metadata):
if metadata.get("id") == vector_id:
return self.vectors[i], metadata
raise ValueError(f"Vector with ID {vector_id} not found")

def delete_by_id(self, vector_id: int) -> bool:
"""Delete vector by ID."""
for i, metadata in enumerate(self.metadata):
if metadata.get("id") == vector_id:
del self.vectors[i]
del self.metadata[i]
return True
return False

def search_by_metadata(self, filter_func) -> List[Tuple[int, Dict]]:
"""Search vectors by metadata filter."""
results = []
for i, metadata in enumerate(self.metadata):
if filter_func(metadata):
results.append((i, metadata))
return results

def save_to_file(self, filename: str):
"""Save database to JSON file."""
data = {
"dimension": self.dimension,
"vectors": [v.tolist() for v in self.vectors],
"metadata": self.metadata,
"id_counter": self.id_counter
}
with open(filename, 'w') as f:
json.dump(data, f, indent=2)

def load_from_file(self, filename: str):
"""Load database from JSON file."""
with open(filename, 'r') as f:
data = json.load(f)

self.dimension = data["dimension"]
self.vectors = [np.array(v) for v in data["vectors"]]
self.metadata = data["metadata"]
self.id_counter = data["id_counter"]

Python - Complete Text Preprocessing Pipeline
# ==========================================
# TEXT PREPROCESSING FOR AI EMBEDDINGS
# ==========================================

import re
import string
from typing import List, Optional

class TextPreprocessor:
"""Complete text preprocessing for AI models."""

def __init__(self):
self.stop_words = {"a", "an", "the", "and", "or", "but", "in", "on", "at"}

def clean_text(self, text: str) -> str:
"""Basic text cleaning."""
# Convert to lowercase
text = text.lower()

# Remove extra whitespace
text = ' '.join(text.split())

# Remove punctuation (optional, keep for some tasks)
# text = text.translate(str.maketrans('', '', string.punctuation))

return text

def tokenize(self, text: str) -> List[str]:
"""Split text into tokens (words)."""
return text.split()

def remove_stop_words(self, tokens: List[str]) -> List[str]:
"""Remove common stop words."""
return [token for token in tokens if token not in self.stop_words]

def stem_tokens(self, tokens: List[str]) -> List[str]:
"""Simple stemming (Porter stemmer in production)."""
stemmed = []
for token in tokens:
# Simple rule-based stemming
if token.endswith('ing'):
token = token[:-3]
elif token.endswith('ed'):
token = token[:-2]
elif token.endswith('s'):
token = token[:-1]
stemmed.append(token)
return stemmed

def create_ngrams(self, tokens: List[str], n: int = 2) -> List[str]:
"""Create n-grams from tokens."""
ngrams = []
for i in range(len(tokens) - n + 1):
ngram = ' '.join(tokens[i:i + n])
ngrams.append(ngram)
return ngrams

def preprocess_pipeline(self, text: str) -> List[str]:
"""Complete preprocessing pipeline."""
# Step 1: Clean text
cleaned = self.clean_text(text)

# Step 2: Tokenize
tokens = self.tokenize(cleaned)

# Step 3: Remove stop words
filtered = self.remove_stop_words(tokens)

# Step 4: Stem tokens
stemmed = self.stem_tokens(filtered)

# Step 5: Create bigrams
bigrams = self.create_ngrams(stemmed, n=2)

# Combine unigrams and bigrams
final_tokens = stemmed + bigrams

return final_tokens

def jaccard_similarity(text1: str, text2: str) -> float:
"""Calculate Jaccard similarity between texts."""
preprocessor = TextPreprocessor()

tokens1 = set(preprocessor.preprocess_pipeline(text1))
tokens2 = set(preprocessor.preprocess_pipeline(text2))

intersection = len(tokens1.intersection(tokens2))
union = len(tokens1.union(tokens2))

return intersection / union if union > 0 else 0

text_pairs = [

("deep learning models", "natural language processing"),
("artificial intelligence", "AI and machine learning")
]

similarity = jaccard_similarity(text1, text2)
print(f"'{text1}' vs '{text2}': {similarity:.3f}")
