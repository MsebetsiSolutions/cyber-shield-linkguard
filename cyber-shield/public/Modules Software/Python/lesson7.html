# 1. Basic while loop with condition
def process_until_target(data, target_value):
"""Process data until we find a value above target."""
index = 0
found = False

while index < len(data) and not found:
if data[index] > target_value:
print(f"🎯 Found at index {index}: {data[index]} > {target_value}")
found = True
index += 1

if not found:
print(f"❌ No value above {target_value} found")

return found

# 3. DANGER ZONE: Infinite loops (and how to escape)
def safe_infinite_monitor():
"""Example of a controlled 'infinite' loop."""
import time

print("⚠️ Starting controlled infinite monitoring...")
print("Press Ctrl+C to stop or wait for timeout\n")

start_time = time.time()
iteration = 0

try:
while True: # This looks infinite, but...
# Safety checks!
if time.time() - start_time > 10: # Timeout after 10 seconds
print("⏰ Safety timeout reached! Exiting...")
break

if iteration > 50: # Maximum iterations
print("🛑 Maximum iterations reached! Exiting...")
break

# Simulate work
iteration += 1
print(f" Iteration {iteration}: Monitoring...", end="\r")
time.sleep(0.2)

except KeyboardInterrupt:
print("\n\n🛑 User interrupted! Exiting safely...")
finally:
print(f"\n✅ Safe exit after {iteration} iterations")

# 1. Working with 2D data (matrix operations)
def analyze_sales_matrix(sales_data):
"""Analyze sales data organized by region and month."""
print("📊 Sales Matrix Analysis")
print("-" * 40)

regions = ["North", "South", "East", "West"]
months = ["Jan", "Feb", "Mar", "Apr"]

# Nested loops for 2D iteration
total_sales = 0
best_region = ""
best_sales = 0

for i, region in enumerate(regions):
region_total = 0
print(f"\n{region} Region:")

for j, month in enumerate(months):
sale = sales_data[i][j]
region_total += sale
total_sales += sale

# Visual indicator for high sales
indicator = "🔥" if sale > 2500 else "✓" if sale > 1500 else "○"
print(f" {month}: ${sale:>6} {indicator}")

# Track best performing region
if region_total > best_sales:
best_sales = region_total
best_region = region

print(f" Region Total: ${region_total}")

print(f"\n📈 Total Company Sales: ${total_sales}")
print(f"🏆 Best Region: {best_region} (${best_sales})")

return total_sales, best_region

# 4. Performance optimization with break/continue
def find_optimal_solution(data, target):
"""Optimized search using break in nested loops."""
print("\n⚡ Optimized Search:")

found = False
operations = 0

for i, row in enumerate(data):
if found: # Early exit if found
break

for j, value in enumerate(row):
operations += 1

if value == target:
print(f" Found {target} at position ({i}, {j})")
print(f" Operations saved: {len(data)*len(row) - operations}")
found = True
break # Exit inner loop

if not found:
print(f" {target} not found in data")

return found

Nested Loop Complexity

Time Complexity: O(n²) for two levels, O(n³) for three levels

Use Case: Matrix operations, combinations, multi-level data

Optimization: Use break/continue, consider vectorization

Real-World Applications

• Sales data by region and month

• Product recommendations (user × item)

• Image processing (pixel × channel)

• Time series analysis (date × metric)

🚀 Nested Loop Optimization Tips:

  1. Minimize inner loop operations: Move calculations outside if possible
  2. Use break/continue effectively: Exit early when condition met
  3. Consider vectorization: Use NumPy for numerical operations
  4. Cache repeated calculations: Store results instead of recomputing
  5. Profile your code: Use timeit to find bottlenecks

💡 Pro Tip: For very large datasets, consider parallel processing or database operations instead of nested Python loops.

# =============================
# DATA ANALYSIS WITH LOOPS
# =============================

class DataAnalyzer:
"""Complete data analysis toolkit using loops."""

def __init__(self, data):
self.data = data
self.analysis_results = {}

def calculate_statistics(self):
"""Calculate comprehensive statistics using loops."""
if not self.data:
return {"error": "No data provided"}

# Initialize variables
total = 0
count = 0
minimum = float('inf')
maximum = float('-inf')

# Single pass through data
for value in self.data:
total += value
count += 1

if value < minimum:
minimum = value

if value > maximum:
maximum = value

# Calculate average
average = total / count if count > 0 else 0

# Calculate variance (second pass)
variance_sum = 0
for value in self.data:
variance_sum += (value - average) ** 2
variance = variance_sum / count if count > 0 else 0

# Store results
self.analysis_results["statistics"] = {
"count": count,
"total": total,
"average": average,
"minimum": minimum,
"maximum": maximum,
"range": maximum - minimum,
"variance": variance,
"std_dev": variance ** 0.5
}

return self.analysis_results["statistics"]

def find_patterns(self, threshold=0.1):
"""Find patterns and trends in the data."""
patterns = {
"increasing_streaks": [],
"decreasing_streaks": [],
"peaks": [],
"valleys": [],
"outliers": []
}

stats = self.analysis_results.get("statistics")
if not stats:
stats = self.calculate_statistics()

current_streak = []
streak_type = None

# Analyze trends and patterns
for i in range(len(self.data)):
current_value = self.data[i]

# Check for outliers
z_score = abs((current_value - stats["average"]) / stats["std_dev"]) if stats["std_dev"] > 0 else 0
if z_score > 2: # More than 2 standard deviations
patterns["outliers"].append({
"index": i,
"value": current_value,
"z_score": z_score
})

# Check for peaks and valleys
if i > 0 and i < len(self.data) - 1:
prev_value = self.data[i - 1]
next_value = self.data[i + 1]

if current_value > prev_value and current_value > next_value:
patterns["peaks"].append({
"index": i,
"value": current_value
})
elif current_value < prev_value and current_value < next_value:
patterns["valleys"].append({
"index": i,
"value": current_value
})

# Track increasing/decreasing streaks
if i > 0:
prev_value = self.data[i - 1]
change = (current_value - prev_value) / prev_value if prev_value != 0 else 0

if change > threshold:
if streak_type != "increasing":
if current_streak:
if streak_type == "increasing":
patterns["increasing_streaks"].append(current_streak.copy())
elif streak_type == "decreasing":
patterns["decreasing_streaks"].append(current_streak.copy())
current_streak = [prev_value, current_value]
streak_type = "increasing"
else:
current_streak.append(current_value)
elif change < -threshold:
if streak_type != "decreasing":
if current_streak:
if streak_type == "increasing":
patterns["increasing_streaks"].append(current_streak.copy())
elif streak_type == "decreasing":
patterns["decreasing_streaks"].append(current_streak.copy())
current_streak = [prev_value, current_value]
streak_type = "decreasing"
else:
current_streak.append(current_value)
else:
# End current streak
if current_streak:
if streak_type == "increasing":
patterns["increasing_streaks"].append(current_streak.copy())
elif streak_type == "decreasing":
patterns["decreasing_streaks"].append(current_streak.copy())
current_streak = []
streak_type = None

# Add final streak if exists
if current_streak:
if streak_type == "increasing":
patterns["increasing_streaks"].append(current_streak.copy())
elif streak_type == "decreasing":
patterns["decreasing_streaks"].append(current_streak.copy())

self.analysis_results["patterns"] = patterns
return patterns

def generate_report(self):
"""Generate a comprehensive analysis report."""
if not self.analysis_results:
self.calculate_statistics()
self.find_patterns()

report = []
report.append("=" * 60)
report.append("DATA ANALYSIS REPORT")
report.append("=" * 60)
report.append(f"Data points: {len(self.data)}")
report.append("")

# Statistics section
stats = self.analysis_results["statistics"]
report.append("📊 STATISTICS")
report.append("-" * 40)
report.append(f"Average: ${stats['average']:.2f}")
report.append(f"Range: ${stats['range']:.2f} (${stats['minimum']:.2f} - ${stats['maximum']:.2f})")
report.append(f"Standard Deviation: ${stats['std_dev']:.2f}")
report.append(f"Total: ${stats['total']:.2f}")
report.append("")

# Patterns section
patterns = self.analysis_results["patterns"]
report.append("🎯 PATTERNS DETECTED")
report.append("-" * 40)
report.append(f"Increasing Streaks: {len(patterns['increasing_streaks'])}")
report.append(f"Decreasing Streaks: {len(patterns['decreasing_streaks'])}")
report.append(f"Peaks: {len(patterns['peaks'])}")
report.append(f"Valleys: {len(patterns['valleys'])}")
report.append(f"Outliers: {len(patterns['outliers'])}")
report.append("")

# Insights section
report.append("💡 INSIGHTS")
report.append("-" * 40)

if stats['average'] > 1000:
report.append("• Strong average performance")

if len(patterns['increasing_streaks']) > len(patterns['decreasing_streaks']):
report.append("• More increasing than decreasing trends")

if patterns['outliers']:
report.append(f"• Found {len(patterns['outliers'])} unusual data points")

report.append("=" * 60)

return "\n".join(report)

# Usage example
sales_data = [1250, 1890, 2100, 1750, 2400, 1980, 3100, 1850, 2200, 1950]
analyzer = DataAnalyzer(sales_data)
print(analyzer.generate_report())

def analyze_with_ai_logic(self, data: List[float]) -> Dict[str, Any]:
"""Perform AI-enhanced analysis using loops."""
print("🤖 AI Analysis Starting...")
print("-" * 40)

# Phase 1: Data preprocessing with loops
print("📥 Phase 1: Data Preprocessing")
cleaned_data = []
for i, value in enumerate(data):
# AI-like cleaning logic

# Simple anomaly detection
if i > 0:
prev_avg = sum(data[max(0, i-3):i]) / min(3, i)

print(f" ⚡ Anomaly detected at index {i}: ${value}")

cleaned_data.append(value)

self.analysis_history.append(results)

print("✅ AI Analysis Complete!")

def _detect_patterns(self, data: List[float]) -> Dict[str, Any]:
"""Advanced pattern detection using nested loops."""
patterns = {
"seasonal_cycles": [],
"trend_changes": [],
"repeating_patterns": [],
"correlation_clusters": []
}

# Look for seasonal cycles (simplified)
print(" Searching for cycles...")
for cycle_length in range(2, min(8, len(data) // 2)):
cycle_score = 0
for i in range(len(data) - cycle_length):
if abs(data[i] - data[i + cycle_length]) / data[i] < 0.1: # Within 10%
cycle_score += 1

if cycle_score > len(data) / 3: # Significant cycle
patterns["seasonal_cycles"].append({
"length": cycle_length,
"confidence": cycle_score / len(data)
})

# Detect trend changes
print(" Analyzing trends...")
window_size = 3
for i in range(window_size, len(data) - window_size):
prev_avg = sum(data[i-window_size:i]) / window_size
next_avg = sum(data[i:i+window_size]) / window_size
change = (next_avg - prev_avg) / prev_avg

if abs(change) > 0.2: # Significant change
patterns["trend_changes"].append({
"index": i,
"change_percent": change * 100,
"direction": "up" if change > 0 else "down"
})

return patterns

def _generate_insights(self, data: List[float], patterns: Dict) -> List[str]:
"""Generate AI-like insights from analysis."""
insights = []

# Calculate basic metrics
avg_value = sum(data) / len(data)
max_value = max(data)
min_value = min(data)

# Generate insights based on data
if avg_value > 2000:
insights.append("💪 Strong overall performance with high average values")

if (max_value - min_value) / avg_value > 0.5:
insights.append("🎢 High volatility detected - consider risk management")

if patterns["seasonal_cycles"]:
cycle_lengths = [c["length"] for c in patterns["seasonal_cycles"]]
insights.append(f"🔄 Seasonal patterns detected at lengths: {cycle_lengths}")

if len(patterns["trend_changes"]) > 2:
insights.append("📈 Multiple trend changes detected - dynamic market conditions")

# Add predictive insight
recent_trend = sum(data[-3:]) / 3 - sum(data[-6:-3]) / 3
if recent_trend > 0:
insights.append("🚀 Recent upward trend suggests continued growth potential")

self.insights.extend(insights)
return insights

def _suggest_optimizations(self, data: List[float]) -> List[str]:
"""Suggest optimizations based on data patterns."""
optimizations = []

# Analyze for optimization opportunities
consecutive_low = 0
for i in range(1, len(data)):
if data[i] < data[i-1]:
consecutive_low += 1
if consecutive_low >= 3:
optimizations.append(f"📉 Consider intervention at position {i}: 3+ consecutive decreases")
else:
consecutive_low = 0

# Check for consistency
avg = sum(data) / len(data)
within_20_percent = sum(1 for x in data if 0.8 * avg <= x <= 1.2 * avg)
consistency = within_20_percent / len(data)

if consistency > 0.8:
optimizations.append(f"🎯 High consistency ({consistency:.0%}) allows for stable planning")
else:
optimizations.append("⚡ Inconsistent data - consider segmentation or clustering")

return optimizations

def _generate_summary(self, data: List[float], patterns: Dict) -> str:
"""Generate a comprehensive AI summary."""
summary_parts = []

summary_parts.append(f"Analyzed {len(data)} data points")
summary_parts.append(f"Average value: ${sum(data)/len(data):.2f}")
summary_parts.append(f"Range: ${min(data):.2f} - ${max(data):.2f}")

if patterns["seasonal_cycles"]:
summary_parts.append(f"Found {len(patterns['seasonal_cycles'])} seasonal patterns")

if patterns["trend_changes"]:
summary_parts.append(f"Detected {len(patterns['trend_changes'])} significant trend changes")

return "; ".join(summary_parts)

# Optimized single-pass implementation
def analyze_sales(data):
if not data:
return {"error": "No data provided"}

total = 0
max_val = data[0]
min_val = data[0]

# Single pass through data
for value in data:
total += value
if value > max_val:
max_val = value
if value < min_val:
min_val = value

avg = total / len(data)

return {
"avg": avg,
"max": max_val,
"min": min_val,
"range": max_val - min_val
}
