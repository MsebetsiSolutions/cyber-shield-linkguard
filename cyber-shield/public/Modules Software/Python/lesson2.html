💻 Part 2: Build Your FastAPI Backend

main.py - Complete FastAPI Application
from fastapi import FastAPI, Request, Form, HTTPException
from fastapi.responses import HTMLResponse, StreamingResponse
from fastapi.templating import Jinja2Templates
from fastapi.staticfiles import StaticFiles
import openai
import os
from dotenv import load_dotenv
import asyncio
from typing import AsyncGenerator

# Load environment variables
load_dotenv()
openai.api_key = os.getenv("OPENAI_API_KEY")

# Initialize FastAPI app
app = FastAPI(
title="AI Assistant API",
description="Your personal AI assistant powered by OpenAI",
version="1.0.0"
)

# Setup templates and static files
templates = Jinja2Templates(directory="templates")
app.mount("/static", StaticFiles(directory="static"), name="static")

async def stream_ai_response(prompt: str) -> AsyncGenerator[str, None]:
"""Stream AI responses like ChatGPT"""
try:
response = openai.ChatCompletion.create(
model="gpt-3.5-turbo",
messages=[
{"role": "system", "content": "You are a helpful AI assistant."},
{"role": "user", "content": prompt}
],
temperature=0.7,
stream=True
)

for chunk in response:
if chunk.choices[0].delta.get("content"):
yield chunk.choices[0].delta.content
await asyncio.sleep(0.01) # Small delay for streaming effect
except Exception as e:
yield f"Error: {str(e)}"

@app.get("/", response_class=HTMLResponse)
async def home(request: Request):
"""Render the main chat interface"""
return templates.TemplateResponse("index.html", {"request": request})

@app.post("/api/chat")
async def chat_endpoint(message: str = Form(...)):
"""Chat endpoint that streams responses"""
return StreamingResponse(
stream_ai_response(message),
media_type="text/event-stream"
)

@app.get("/api/health")
async def health_check():
"""Health check endpoint for deployment"""
return {"status": "healthy", "service": "AI Assistant API"}

if __name__ == "__main__":
import uvicorn
uvicorn.run(app, host="0.0.0.0", port=8000)
ai_assistant.py - Enhanced from Lesson 1
import openai
import os
from dotenv import load_dotenv

class AIAssistant:
"""Enhanced AI Assistant with streaming support"""

def __init__(self):
load_dotenv()
self.api_key = os.getenv("OPENAI_API_KEY")
openai.api_key = self.api_key

if not self.api_key:
raise ValueError("OPENAI_API_KEY not found in .env file")

def stream_response(self, prompt: str):
"""Generator that yields AI response in chunks"""
try:
response = openai.ChatCompletion.create(
model="gpt-3.5-turbo",
messages=[
{"role": "system", "content": "You are a helpful AI assistant."},
{"role": "user", "content": prompt}
],
temperature=0.7,
stream=True,
max_tokens=500
)

for chunk in response:
if chunk.choices[0].delta.get("content"):
yield chunk.choices[0].delta.content
except Exception as e:
yield f"Error: {str(e)}"

def get_response(self, prompt: str) -> str:
"""Get complete AI response (non-streaming)"""
try:
response = openai.ChatCompletion.create(
model="gpt-3.5-turbo",
messages=[
{"role": "system", "content": "You are a helpful AI assistant."},
{"role": "user", "content": prompt}
],
temperature=0.7,
max_tokens=500
)
return response.choices[0].message.content
except Exception as e:
return f"Error: {str(e)}"

☁️ Part 3: Deploy to the Cloud (FREE)

Step-by-Step Deployment to Render.com

1
Create Render Account

Go to render.com and sign up with GitHub (free)

2
Create render.yaml File

Add this configuration to your project:

render.yaml
services:
- type: web
name: ai-assistant
env: python
buildCommand: pip install -r requirements.txt
startCommand: uvicorn main:app --host 0.0.0.0 --port 10000
envVars:
- key: OPENAI_API_KEY
sync: false
3
Push to GitHub

git add .
git commit -m "Add AI web application"
git push origin main

4
Deploy on Render

1. Click "New +" on Render
2. Select "Blueprint"
3. Connect your GitHub repo
4. Add your OPENAI_API_KEY in Environment
5. Click "Create Blueprint"

5
Get Your Live URL
