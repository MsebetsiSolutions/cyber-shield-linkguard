💻 Build Your AI Assistant - Step by Step

Step 1: Create Your AI Assistant File

ai_assistant.py
# Your First AI Assistant - Complete Code
import openai
import os
from dotenv import load_dotenv

# Load your API key safely
load_dotenv()
openai.api_key = os.getenv("OPENAI_API_KEY")

def ask_ai(question):
"""Talk to an AI assistant"""
try:
response = openai.ChatCompletion.create(
model="gpt-3.5-turbo",
messages=[
{"role": "system", "content": "You are a helpful Python programming assistant."},
{"role": "user", "content": question}
],
temperature=0.7
)
return response.choices[0].message.content
except Exception as e:
return f"Error: {str(e)}"

# Main program
if __name__ == "__main__":
print("🤖 AI Assistant Ready! Type 'quit' to exit.")
while True:
question = input("\\nYou: ")
if question.lower() == "quit":
break
answer = ask_ai(question)
print(f"AI: {answer}")

# Save conversation to file
with open("chat_history.txt", "a") as f:
f.write(f"You: {question}\\nAI: {answer}\\n\\n")
