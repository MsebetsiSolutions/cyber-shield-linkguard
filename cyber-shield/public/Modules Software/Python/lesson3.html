🏗️ OOP for AI Systems (OpenAI Architecture)

OpenAI uses OOP to organize complex AI systems. Here's a simplified version of their architecture:

Python
# OpenAI-style AI System Architecture
class AIConversation:
"""Represents a single AI conversation"""

def __init__(self, system_prompt="You are a helpful assistant."):
self.messages = [{"role": "system", "content": system_prompt}]
self.token_count = 0

def add_message(self, role, content):
"""Add a message to the conversation"""
self.messages.append({"role": role, "content": content})
self.token_count += len(content.split()) # Simplified token count

def get_conversation_history(self):
"""Return formatted conversation history"""
return self.messages

def __str__(self):
return f"AIConversation with {len(self.messages)} messages ({self.token_count} tokens)"

class AIAssistant:
"""Main AI assistant class (like ChatGPT's backend)"""

def __init__(self, model="gpt-3.5-turbo"):
self.model = model
self.conversations = []

def start_new_conversation(self, system_prompt=None):
"""Start a new conversation"""
conv = AIConversation(system_prompt or "You are a helpful assistant.")
self.conversations.append(conv)
return conv

def respond_to_message(self, conversation, user_message):
"""Process user message and generate AI response"""
conversation.add_message("user", user_message)

# Simulate AI response generation
ai_response = self._generate_response(user_message)
conversation.add_message("assistant", ai_response)

return ai_response

def _generate_response(self, user_message):
"""Simulate AI response generation"""
responses = [
f"I understand you're asking: {user_message}",
"That's an interesting question!",
"Let me help you with that.",
"Based on your query, here's what I think..."
]
return random.choice(responses)
