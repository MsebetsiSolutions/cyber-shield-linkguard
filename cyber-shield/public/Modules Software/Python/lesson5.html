from typing import List, Dict, Optional
from dataclasses import dataclass
from datetime import datetime
import openai

@dataclass
class AIResponse:
"""Data model for AI responses."""
content: str
model: str
tokens: int
timestamp: datetime

class AIService:
"""Service class for OpenAI API calls."""

def __init__(self, api_key: str, default_model: str = "gpt-3.5-turbo"):
self.api_key = api_key
self.default_model = default_model
openai.api_key = api_key
self.conversation_history: List[Dict] = []

def generate_response(self, prompt: str, **kwargs) -> AIResponse:
"""Generate AI response with conversation context."""
try:
response = openai.ChatCompletion.create(
model=kwargs.get('model', self.default_model),
messages=self.conversation_history + [
{"role": "user", "content": prompt}
],
**kwargs
)

ai_response = AIResponse(
content=response.choices[0].message.content,
model=response.model,
tokens=response.usage.total_tokens,
timestamp=datetime.now()
)

# Update conversation history
self.conversation_history.append({"role": "user", "content": prompt})
self.conversation_history.append({"role": "assistant", "content": ai_response.content})

return ai_response

except Exception as e:
raise AIServiceError(f"API call failed: {str(e)}")

def clear_history(self) -> None:
"""Clear conversation history."""
self.conversation_history = []

class AIServiceError(Exception):
"""Custom exception for AI service errors."""
pass

Python - Flask AI App
from flask import Flask, request, jsonify
from flask_cors import CORS
import os
from dotenv import load_dotenv

# Load our AIService class from earlier
from ai_service import AIService

load_dotenv()

app = Flask(__name__)
CORS(app) # Enable Cross-Origin requests

# Initialize our AI service
ai_service = AIService(api_key=os.getenv("OPENAI_API_KEY"))

@app.route('/api/chat', methods=['POST'])
def chat():
"""Chat endpoint that uses our AIService class."""
try:
data = request.get_json()
message = data.get('message', '')

if not message:
return jsonify({"error": "Message is required"}), 400

# Use our OOP service
response = ai_service.generate_response(message)

return jsonify({
"success": True,
"response": response.content,
"tokens": response.tokens,
"model": response.model
})

except Exception as e:
return jsonify({"error": str(e)}), 500

@app.route('/api/clear', methods=['POST'])
def clear_chat():
"""Clear conversation history."""
ai_service.clear_history()
return jsonify({"success": True})

if __name__ == '__main__':
app.run(debug=True, port=5000)
Python - FastAPI AI App
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from typing import Optional
import os
from dotenv import load_dotenv

from ai_service import AIService

load_dotenv()

app = FastAPI(title="AI Chat API", version="1.0.0")

# CORS middleware
app.add_middleware(
CORSMiddleware,
allow_origins=["*"],
allow_methods=["*"],
allow_headers=["*"],
)

# Pydantic models for type validation
class ChatRequest(BaseModel):
message: str
model: Optional[str] = None
temperature: Optional[float] = 0.7

class ChatResponse(BaseModel):
success: bool
response: str
tokens: int
model: str

# Initialize service
ai_service = AIService(api_key=os.getenv("OPENAI_API_KEY"))

@app.post("/api/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
"""Async chat endpoint with automatic documentation."""
try:
response = ai_service.generate_response(
request.message,
model=request.model,
temperature=request.temperature
)

return ChatResponse(
success=True,
response=response.content,
tokens=response.tokens,
model=response.model
)

except Exception as e:
raise HTTPException(status_code=500, detail=str(e))

@app.post("/api/clear")
async def clear_chat():
"""Clear conversation history."""
ai_service.clear_history()
return {"success": True}

# Run: uvicorn main:app --reload

🤖 Advanced AI Service Classes

Let's build a multi-provider AI orchestrator that can switch between OpenAI, Anthropic, and Cohere:

Python - Multi-AI Orchestrator
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
from dataclasses import dataclass
from enum import Enum
import openai
import anthropic
import cohere

class AIModel(Enum):
"""Available AI models."""
GPT_35_TURBO = "gpt-3.5-turbo"
GPT_4 = "gpt-4"
CLAUDE_2 = "claude-2"
CLAUDE_3 = "claude-3-opus"
COMMAND = "command"

class AIProvider(ABC):
"""Abstract base class for AI providers."""

@abstractmethod
def generate(self, prompt: str, **kwargs) -> Dict[str, Any]:
pass

@abstractmethod
def get_cost(self, tokens: int) -> float:
pass

class OpenAIService(AIProvider):
"""OpenAI implementation."""

def __init__(self, api_key: str):
self.client = openai.OpenAI(api_key=api_key)

def generate(self, prompt: str, model: AIModel = AIModel.GPT_35_TURBO, **kwargs):
response = self.client.chat.completions.create(
model=model.value,
messages=[{"role": "user", "content": prompt}],
**kwargs
)
return {
"content": response.choices[0].message.content,
"tokens": response.usage.total_tokens,
"model": response.model
}

def get_cost(self, tokens: int) -> float:
return tokens * 0.002 / 1000 # $0.002 per 1K tokens

class AIOrchestrator:
"""Orchestrates multiple AI providers with failover."""

def __init__(self):
self.providers: Dict[str, AIProvider] = {}
self.default_provider = "openai"

def add_provider(self, name: str, provider: AIProvider):
self.providers[name] = provider

def generate_with_failover(self, prompt: str, **kwargs) -> Dict[str, Any]:
"""Try providers in order until one works."""
for name, provider in self.providers.items():
try:
result = provider.generate(prompt, **kwargs)
result["provider"] = name
return result
except Exception as e:
print(f"Provider {name} failed: {e}")
continue
raise Exception("All AI providers failed")
