# 🎓 MASTERCLASS: Advanced Prompt Engineering System
from typing import Dict, List, Any, Optional, Callable
import re
import json
from datetime import datetime
from dataclasses import dataclass, asdict
from enum import Enum
import asyncio
from functools import wraps

class PromptEngineeringLevel(Enum):
    BASIC = "basic"
    INTERMEDIATE = "intermediate"
    ADVANCED = "advanced"
    EXPERT = "expert"

@dataclass
class PromptTemplate:
    """Professional prompt template for manufacturing AI"""
    name: str
    template: str
    variables: List[str]
    level: PromptEngineeringLevel
    description: str
    expected_output_format: str
    validation_rules: Dict[str, Any]
    
    def render(self, **kwargs) -> str:
        """Render template with provided variables"""
        try:
            rendered = self.template
            for key, value in kwargs.items():
                placeholder = f"{{{key}}}"
                if placeholder in rendered:
                    rendered = rendered.replace(placeholder, str(value))
            
            # Validate all placeholders are filled
            unfilled = re.findall(r'\{(\w+)\}', rendered)
            if unfilled:
                raise ValueError(f"Unfilled variables: {unfilled}")
            
            return rendered
        except Exception as e:
            raise ValueError(f"Failed to render template: {str(e)}")
