    def get_total_personnel(self):
        """Calculate total active personnel using list comprehension"""
        # YOUR CODE HERE: Use list comprehension to count active personnel
        active_personnel = [p for p in self.personnel if p["active"]]
        return len(active_personnel)
    
    def get_personnel_list(self):
        """Get formatted personnel list using list comprehension"""
        # YOUR CODE HERE: Use list comprehension to create formatted strings
        personnel_list = [f"{p['access_id']}: {p['name']} ({p['role']})" 
                         for p in self.personnel if p["active"]]
        return personnel_list
    
    def get_zone_occupancy(self, zone):
        """Get personnel in specific zone using list comprehension"""
        # YOUR CODE HERE: Use list comprehension to filter by zone access
        zone_personnel = [p["name"] for p in self.personnel 
                         if p["active"] and zone in p["zone_access"]]
        return zone_personnel
    
    def get_skills_inventory(self):
        """Get all unique skills using nested list comprehension"""
        # YOUR CODE HERE: Use nested list comprehension to flatten skills
        all_skills = [skill for p in self.personnel for skill in p["skills"]]
        return list(set(all_skills))  # Remove duplicates
    
    def get_role_summary(self):
        """Get count by role using dictionary comprehension"""
        # YOUR CODE HERE: Use dictionary comprehension to count by role
        role_counts = {}
        roles = [p["role"] for p in self.personnel if p["active"]]
        for role in set(roles):
            role_counts[role] = roles.count(role)
        return role_counts

class ManufacturingDataProcessor:
    """Advanced comprehensions for manufacturing data"""
    
    @staticmethod
    def process_sensor_readings(raw_data):
        """Dictionary comprehension example"""
        # Convert raw sensor data to structured format
        sensor_data = {
            sensor_id: {
                'value': float(value),
                'unit': '°C' if 'temp' in sensor_id else 'psi',
                'status': 'normal' if float(value) < 80 else 'warning'
            }
            for sensor_id, value in raw_data.items()
            if value.replace('.', '').isdigit()
        }
        return sensor_data
    
    @staticmethod
    def generate_production_batch(batch_size):
        """Generator comprehension example"""
        # Generate large dataset without storing in memory
        return (
            {
                'batch_id': f'BATCH_{i:06d}',
                'units': i * 100,
                'quality_score': min(100, 95 + (i % 10)),
                'energy_used': i * 1.5
            }
            for i in range(batch_size)
            if i % 2 == 0  # Only even batches
        )
    
    @staticmethod
    def analyze_defect_patterns(defect_reports):
        """Set comprehension example"""
        # Extract unique defect types and locations
        unique_defects = {
            (report['type'], report['location'])
            for report in defect_reports
            if report.get('type') and report.get('location')
        }
        
        # Count defects by type using dictionary comprehension
        defect_counts = {
            defect_type: sum(1 for r in defect_reports if r['type'] == defect_type)
            for defect_type in set(r['type'] for r in defect_reports)
        }
        
        return unique_defects, defect_counts
    
    @staticmethod
    def optimize_production_schedule(production_data):
        """Complex nested comprehensions"""
        # Calculate efficiency metrics for each line
        efficiency_metrics = {
            line['id']: {
                'oee': line['good_units'] / line['planned_units'],
                'utilization': line['runtime'] / line['planned_runtime'],
                'quality': line['good_units'] / (line['good_units'] + line['defects'])
            }
            for line in production_data
            if line['planned_units'] > 0
        }
        
        # Filter high-performance lines
        high_perf_lines = [
            line_id for line_id, metrics in efficiency_metrics.items()
            if all(v > 0.85 for v in metrics.values())
        ]
        
        return efficiency_metrics, high_perf_lines

# 📄 database.py - Manufacturing data models
import os
import psycopg2
from datetime import datetime
from contextlib import contextmanager

class ManufacturingDatabase:
    def __init__(self):
        self.db_url = os.getenv('DATABASE_URL')
    
    @contextmanager
    def get_connection(self):
        conn = psycopg2.connect(self.db_url)
        try:
            yield conn
        finally:
            conn.close()
    
    def log_production_event(self, line_id, event_type, details):
        """Log manufacturing events to database"""
        with self.get_connection() as conn:
            with conn.cursor() as cur:
                cur.execute("""
                    INSERT INTO production_events 
                    (line_id, event_type, details, timestamp)
                    VALUES (%s, %s, %s, %s)
                    RETURNING id
                """, (line_id, event_type, details, datetime.utcnow()))
                conn.commit()
                return cur.fetchone()[0]
    
    def get_production_stats(self, hours=24):
        """Get production statistics for dashboard"""
        with self.get_connection() as conn:
            with conn.cursor() as cur:
                cur.execute("""
                    SELECT 
                        COUNT(*) as total_events,
                        COUNT(DISTINCT line_id) as active_lines,
                        AVG(CASE WHEN event_type = 'defect' THEN 1 ELSE 0 END) as defect_rate,
                        MAX(timestamp) as last_update
                    FROM production_events
                    WHERE timestamp >= NOW() - INTERVAL '%s hours'
                """, (hours,))
                return dict(zip([desc[0] for desc in cur.description], cur.fetchone()))

import os
import json
from datetime import datetime
import logging
from typing import Dict, Any

class ManufacturingAIMonitor:
    """Production monitoring for manufacturing AI applications"""
    
    def __init__(self):
        # Load environment variables
        self.api_key = os.getenv('MANUFACTURING_API_KEY')
        self.model_version = os.getenv('MODEL_VERSION', 'v1.0')
        self.debug_mode = os.getenv('DEBUG_MODE', 'False').lower() == 'true'
        
        # Setup logging
        log_level = os.getenv('LOG_LEVEL', 'INFO')
        logging.basicConfig(
            level=getattr(logging, log_level),
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        self.logger = logging.getLogger('manufacturing_ai')
        
        # Initialize metrics
        self.metrics = {
            'requests_processed': 0,
            'errors': 0,
            'avg_response_time': 0,
            'predictions_made': 0,
            'start_time': datetime.now()
        }
    
    def process_manufacturing_request(self, request_data: Dict[str, Any]) -> Dict[str, Any]:
        """Process manufacturing AI request with monitoring"""
        start_time = datetime.now()
        
        try:
            # Log request
            self.logger.info(f"Processing request: {request_data.get('type', 'unknown')}")
            
            # Validate API key
            if not self.validate_api_key():
                raise ValueError("Invalid API key")
            
            # Process based on request type
            if request_data.get('type') == 'predictive_maintenance':
                result = self.predict_maintenance(request_data)
            elif request_data.get('type') == 'quality_inspection':
                result = self.inspect_quality(request_data)
            else:
                result = self.process_general(request_data)
            
            # Update metrics
            self.update_metrics(success=True, duration=(datetime.now() - start_time).total_seconds())
            
            return {
                'success': True,
                'result': result,
                'model_version': self.model_version,
                'processing_time': (datetime.now() - start_time).total_seconds()
            }
            
        except Exception as e:
            self.logger.error(f"Error processing request: {str(e)}")
            self.update_metrics(success=False)
            
            return {
                'success': False,
                'error': str(e),
                'model_version': self.model_version
            }
    
    def validate_api_key(self) -> bool:
        """Validate manufacturing API key"""
        # In production, validate against your authentication service
        return bool(self.api_key and len(self.api_key) > 10)
    
    def predict_maintenance(self, data: Dict) -> Dict:
        """Predict maintenance needs for manufacturing equipment"""
        # Your AI model logic here
        return {
            'maintenance_needed': True,
            'confidence': 0.92,
            'recommended_action': 'Schedule maintenance in 48 hours',
            'critical_components': ['bearing_assembly', 'hydraulic_pump']
        }
    
    def update_metrics(self, success: bool, duration: float = 0):
        """Update monitoring metrics"""
        self.metrics['requests_processed'] += 1
        
        if not success:
            self.metrics['errors'] += 1
        
        # Update average response time
        old_avg = self.metrics['avg_response_time']
        old_count = self.metrics['requests_processed'] - 1
        self.metrics['avg_response_time'] = (old_avg * old_count + duration) / self.metrics['requests_processed']
        
        if success and duration > 0:
            self.metrics['predictions_made'] += 1
    
    def get_health_status(self) -> Dict:
        """Get application health status"""
        uptime = (datetime.now() - self.metrics['start_time']).total_seconds()
        
        return {
            'status': 'healthy',
            'uptime_seconds': uptime,
            'metrics': self.metrics,
            'environment': {
                'model_version': self.model_version,
                'debug_mode': self.debug_mode,
                'log_level': self.logger.level
            }
        }
    
    def generate_alert(self, alert_type: str, message: str):
        """Generate production alert"""
        alert = {
            'type': alert_type,
            'message': message,
            'timestamp': datetime.now().isoformat(),
            'metrics': self.metrics.copy()
        }
        
        # Log alert
        self.logger.warning(f"Alert: {alert_type} - {message}")
        
        # In production, send to Slack/Teams/PagerDuty
        webhook_url = os.getenv('ALERT_WEBHOOK')
        if webhook_url and not self.debug_mode:
            # Send to webhook (simplified)
            self.logger.info(f"Alert sent to webhook: {alert_type}")
        
        return alert
