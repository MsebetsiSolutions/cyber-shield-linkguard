) -> List[int]:

Args:
  data: List of input arrays for prediction
  model: Trained AI model for inference
  threshold: Confidence threshold for classification

Returns:
  List of binary predictions (0 or 1)

  1 if model.predict(d) > threshold else 0

📡 Make Your First AI API Call

Let's write a clean function to call the OpenAI API. This is the foundation of all AI applications.

Python
import os
from typing import Optional, Dict, Any
from dotenv import load_dotenv
import openai

# Load environment variables
load_dotenv()

def call_openai(
prompt: str,
model: str = "gpt-3.5-turbo",
temperature: float = 0.7,
max_tokens: Optional[int] = None,
system_message: str = "You are a helpful AI assistant."
) -> Dict[str, Any]:
"""
Call OpenAI's Chat Completions API with proper error handling.

Args:
prompt: User's message to the AI
model: Which OpenAI model to use
temperature: Creativity control (0.0-1.0)
max_tokens: Maximum response length
system_message: AI's role/personality

Returns:
Dictionary with 'success', 'response', and 'usage' keys
"""
try:
response = openai.ChatCompletion.create(
model=model,
messages=[
{"role": "system", "content": system_message},
{"role": "user", "content": prompt}
],
temperature=temperature,
max_tokens=max_tokens
)

return {
"success": True,
"response": response.choices[0].message.content,
"usage": response.usage.dict(),
"model": response.model
}

except openai.error.AuthenticationError:
return {"success": False, "error": "Invalid API key"}
except openai.error.RateLimitError:
return {"success": False, "error": "Rate limit exceeded"}
except Exception as e:
return {"success": False, "error": str(e)}

"""
AI Chatbot - Complete Implementation
Combines clean functions, type hints, and API calls
"""

import os
from typing import List, Dict, Optional
from dataclasses import dataclass, asdict
from datetime import datetime
from dotenv import load_dotenv
import openai

load_dotenv()
openai.api_key = os.getenv("OPENAI_API_KEY")

@dataclass
class ChatMessage:
"""Data class for chat messages."""
role: str # 'user' or 'assistant'
content: str
timestamp: datetime

class AIChatbot:
"""Main chatbot class with conversation history."""

def __init__(
self,
system_prompt: str = "You are a helpful AI assistant.",
model: str = "gpt-3.5-turbo"
) -> None:
self.system_prompt = system_prompt
self.model = model
self.conversation_history: List[ChatMessage] = []
self._initialize_conversation()

def _initialize_conversation(self) -> None:
"""Add system message to start conversation."""
system_msg = ChatMessage(
role="system",
content=self.system_prompt,
timestamp=datetime.now()
)
self.conversation_history.append(system_msg)

def add_user_message(self, content: str) -> None:
"""Add user message to conversation history."""
message = ChatMessage(
role="user",
content=content,
timestamp=datetime.now()
)
self.conversation_history.append(message)

def get_ai_response(self) -> Dict[str, any]:
"""Get AI response based on conversation history."""
try:
response = openai.ChatCompletion.create(
model=self.model,
messages=[
                asdict(msg) for msg in self.conversation_history
],
temperature=0.7,
max_tokens=500
)

ai_content = response.choices[0].message.content
ai_message = ChatMessage(
role="assistant",
content=ai_content,
timestamp=datetime.now()
)
self.conversation_history.append(ai_message)

return {
"success": True,
"response": ai_content,
"tokens": response.usage.total_tokens
}

except Exception as e:
return {"success": False, "error": str(e)}

def chat_loop(self) -> None:
"""Run interactive chat loop."""
print("🤖 AI Chatbot Started! Type 'quit' to exit.")

while True:
user_input = input("\\nYou: ").strip()

if user_input.lower() == "quit":
print("Goodbye! 👋")
break

self.add_user_message(user_input)
result = self.get_ai_response()

if result["success"]:
print(f"🤖 AI: {result['response']}")
print(f"📊 Tokens used: {result['tokens']}")
else:
print(f"❌ Error: {result['error']}")
