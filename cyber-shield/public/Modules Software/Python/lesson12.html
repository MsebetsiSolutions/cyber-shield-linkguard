class ManufacturingDataProcessor:
    """Process manufacturing data using string methods"""
    
    def clean_sensor_data(self, raw_data):
        """Clean and standardize sensor readings"""
        cleaned = {}
        
        for sensor_id, reading in raw_data.items():
            # Remove whitespace and convert to consistent format
            clean_id = sensor_id.strip().replace(' ', '_').upper()
            
            # Handle different reading formats
            if isinstance(reading, str):
                clean_reading = reading.strip().replace('°C', '').replace('C', '')
                if clean_reading.replace('.', '').isdigit():
                    cleaned[clean_id] = float(clean_reading)
                else:
                    cleaned[clean_id] = clean_reading
            else:
                cleaned[clean_id] = reading
        
        return cleaned
    
    def validate_equipment_id(self, equipment_id):
        """Validate manufacturing equipment IDs"""
        rules = [
            equipment_id.isalnum() or '_' in equipment_id,
            len(equipment_id) <= 20,
            not equipment_id.startswith('_'),
            not equipment_id.endswith('_')
        ]
        
        return all(rules)
    
    def extract_equipment_info(self, equipment_string):
        """Extract type, model, and serial from equipment string"""
        # Example: "CNC_MILL_HAAS_VF2_SN_12345"
        parts = equipment_string.split('_')
        
        info = {
            'type': parts[0] if len(parts) > 0 else '',
            'model': '_'.join(parts[1:-2]) if len(parts) > 2 else '',
            'serial': parts[-1] if parts[-1].isdigit() else ''
        }
        
        return info

import random
from datetime import datetime, timedelta

class ProductionDataManager:
    """Manage production data using list methods"""
    
    def __init__(self):
        self.production_batches = []
        self.quality_scores = []
        self.downtime_records = []
    
    def generate_production_data(self, days=30):
        """Generate simulated production data"""
        start_date = datetime.now() - timedelta(days=days)
        
        for i in range(days):
            date = start_date + timedelta(days=i)
            
            # Simulate production metrics
            batch = {
                'date': date.date(),
                'units_produced': random.randint(800, 1200),
                'defect_count': random.randint(0, 20),
                'downtime_minutes': random.randint(0, 120),
                'energy_consumption': random.uniform(800, 1200)
            }
            
            self.production_batches.append(batch)
            self.quality_scores.append(1 - (batch['defect_count'] / batch['units_produced']))
        
        return self.production_batches
    
    def analyze_production_trends(self):
        """Analyze production trends using list methods"""
        if not self.production_batches:
            self.generate_production_data()
        
        # Extract metrics using list comprehensions
        units = [batch['units_produced'] for batch in self.production_batches]
        defects = [batch['defect_count'] for batch in self.production_batches]
        downtimes = [batch['downtime_minutes'] for batch in self.production_batches]
        
        # Calculate statistics
        analysis = {
            'total_units': sum(units),
            'avg_daily_units': sum(units) / len(units),
            'max_units': max(units),
            'min_units': min(units),
            'total_defects': sum(defects),
            'defect_rate': sum(defects) / sum(units),
            'total_downtime': sum(downtimes),
            'worst_day': self.production_batches[downtimes.index(max(downtimes))]['date']
        }
        
        return analysis
    
    def optimize_production_schedule(self):
        """Optimize production using list sorting and filtering"""
        # Sort by efficiency (units/downtime)
        sorted_batches = sorted(
            self.production_batches,
            key=lambda x: x['units_produced'] / (x['downtime_minutes'] + 1),
            reverse=True
        )
        
        # Filter high-performance days
        high_perf_days = [
            batch for batch in sorted_batches[:5]
            if batch['defect_count'] < 10 and batch['downtime_minutes'] < 30
        ]
        
        # Find patterns in high-performance days
        patterns = {
            'avg_units_high_perf': sum(b['units_produced'] for b in high_perf_days) / len(high_perf_days),
            'common_downtime_causes': self.analyze_downtime_patterns(high_perf_days)
        }
        
        return sorted_batches, high_perf_days, patterns
    
    def analyze_downtime_patterns(self, batches):
        """Analyze downtime patterns using list methods"""
        # Group by downtime ranges
        downtime_ranges = {
            'low': [b for b in batches if b['downtime_minutes'] < 30],
            'medium': [b for b in batches if 30 <= b['downtime_minutes'] < 90],
            'high': [b for b in batches if b['downtime_minutes'] >= 90]
        }
        
        return downtime_ranges

import json

import statistics

class IoTSensorManager:
    """Manage IoT sensor data using dictionary methods"""
    
    def __init__(self):
        self.sensor_network = defaultdict(dict)
        self.sensor_readings = []
        self.alerts = []
    
    def add_sensor(self, sensor_id, sensor_type, location, config=None):
        """Add a new sensor to the network"""
        sensor_data = {
            'type': sensor_type,
            'location': location,
            'status': 'active',
            'config': config or {},
            'readings': [],
            'last_updated': None
        }
        
        self.sensor_network[sensor_id] = sensor_data
        return sensor_id
    
    def process_sensor_data(self, raw_data):
        """Process incoming sensor data"""
        processed = {}
        
        for sensor_id, reading in raw_data.items():
            if sensor_id in self.sensor_network:
                # Get sensor configuration
                config = self.sensor_network[sensor_id].get('config', {})
                
                # Process based on sensor type
                sensor_type = self.sensor_network[sensor_id]['type']
                
                if sensor_type == 'temperature':
                    processed_value = self.process_temperature(reading, config)
                elif sensor_type == 'vibration':
                    processed_value = self.process_vibration(reading, config)
                elif sensor_type == 'pressure':
                    processed_value = self.process_pressure(reading, config)
                else:
                    processed_value = reading
                
                # Update sensor record
                self.sensor_network[sensor_id]['readings'].append(processed_value)
                self.sensor_network[sensor_id]['last_updated'] = 'now'
                
                # Check for anomalies
                if self.detect_anomaly(sensor_id, processed_value):
                    self.create_alert(sensor_id, processed_value)
                
                processed[sensor_id] = processed_value
        
        self.sensor_readings.append(processed)
        return processed
    
    def process_temperature(self, reading, config):
        """Process temperature sensor data"""
        # Convert to Celsius if needed
        if isinstance(reading, str) and 'F' in reading:
            temp_f = float(reading.replace('F', '').strip())
            return (temp_f - 32) * 5/9
        return float(reading)
    
    def detect_anomaly(self, sensor_id, value):
        """Detect anomalies in sensor readings"""
        readings = self.sensor_network[sensor_id]['readings']
        
        if len(readings) > 10:
            recent = readings[-10:]
            mean = statistics.mean(recent)
            stdev = statistics.stdev(recent) if len(recent) > 1 else 0
            
            # Check if value is outside 3 standard deviations
            if stdev > 0 and abs(value - mean) > 3 * stdev:
                return True
        
        return False
    
    def create_alert(self, sensor_id, value):
        """Create anomaly alert"""
        alert = {
            'sensor_id': sensor_id,
            'value': value,
            'type': 'anomaly',
            'timestamp': 'now',
            'severity': 'high' if abs(value) > 100 else 'medium'
        }
        
        self.alerts.append(alert)
        return alert
    
    def get_sensor_stats(self):
        """Get statistics for all sensors"""
        stats = {}
        
        for sensor_id, data in self.sensor_network.items():
            readings = data['readings']
            if readings:
                stats[sensor_id] = {
                    'count': len(readings),
                    'mean': statistics.mean(readings) if readings else 0,
                    'min': min(readings) if readings else 0,
                    'max': max(readings) if readings else 0,
                    'last_value': readings[-1] if readings else None
                }
        
        return stats
    
    def find_correlated_sensors(self):
        """Find sensors with correlated readings"""
        correlations = {}
        
        sensor_ids = list(self.sensor_network.keys())
        
        for i, sensor_a in enumerate(sensor_ids):
            readings_a = self.sensor_network[sensor_a]['readings']
            
            for sensor_b in sensor_ids[i+1:]:
                readings_b = self.sensor_network[sensor_b]['readings']
                
                # Simple correlation calculation
                if len(readings_a) == len(readings_b) and len(readings_a) > 5:
                    try:
                        corr = statistics.correlation(readings_a, readings_b)
                        if abs(corr) > 0.7:  # Strong correlation
                            correlations[(sensor_a, sensor_b)] = corr
                    except:
                        pass
        
        return correlations

import numpy as np

class ManufacturingCorrelationAnalyzer:
    """Analyze correlations in manufacturing data"""
    
    def __init__(self):
        self.data = pd.DataFrame()
    
    def generate_manufacturing_data(self, n_samples=100):
        """Generate synthetic manufacturing data with correlations"""
        np.random.seed(42)
        
        # Create correlated features
        # Production rate influences quality and energy consumption
        production_rate = np.random.normal(100, 15, n_samples)
        
        # Quality negatively correlated with defect rate
        quality_score = 95 - 0.3 * np.abs(np.random.normal(0, 5, n_samples)) + 0.1 * production_rate
        
        # Energy consumption increases with production
        energy_kwh = 800 + 2 * production_rate + np.random.normal(0, 50, n_samples)
        
        # Downtime has inverse relationship with maintenance score
        maintenance_score = np.random.uniform(70, 100, n_samples)
        downtime_hours = 50 - 0.4 * maintenance_score + np.random.normal(0, 10, n_samples)
        
        # Safety incidents (should be low)
        safety_incidents = np.random.poisson(0.5, n_samples)
        
        # Create DataFrame
        self.data = pd.DataFrame({
            'production_rate': production_rate,
            'quality_score': quality_score,
            'energy_kwh': energy_kwh,
            'downtime_hours': downtime_hours,
            'maintenance_score': maintenance_score,
            'safety_incidents': safety_incidents
        })
        
        return self.data
    
    def calculate_correlations(self):
        """Calculate correlation matrix"""
        if self.data.empty:
            self.generate_manufacturing_data()
        
        correlation_matrix = self.data.corr()
        return correlation_matrix
    
    def visualize_correlations(self):
        """Create heatmap visualization of correlations"""
        corr_matrix = self.calculate_correlations()
        
        fig, ax = plt.subplots(figsize=(10, 8))
        
        # Create heatmap
        im = ax.imshow(corr_matrix.values, cmap='RdYlBu', vmin=-1, vmax=1)
        
        # Set labels
        ax.set_xticks(np.arange(len(corr_matrix.columns)))
        ax.set_yticks(np.arange(len(corr_matrix.columns)))
        ax.set_xticklabels(corr_matrix.columns, rotation=45, ha='right')
        ax.set_yticklabels(corr_matrix.columns)
        
        # Add correlation values
        for i in range(len(corr_matrix.columns)):
            for j in range(len(corr_matrix.columns)):
                text = ax.text(j, i, f'{corr_matrix.iloc[i, j]:.2f}',
                             ha="center", va="center", color="black", fontsize=10)
        
        ax.set_title("🏭 Manufacturing Metrics Correlation Matrix", fontsize=14, fontweight='bold')
        plt.colorbar(im)
        plt.tight_layout()
        
        return fig
    
    def find_strong_correlations(self, threshold=0.7):
        """Find strongly correlated metric pairs"""
        corr_matrix = self.calculate_correlations()
        
        strong_correlations = []
        
        for i in range(len(corr_matrix.columns)):
            for j in range(i+1, len(corr_matrix.columns)):
                corr_value = corr_matrix.iloc[i, j]
                if abs(corr_value) >= threshold:
                    strong_correlations.append({
                        'metric1': corr_matrix.columns[i],
                        'metric2': corr_matrix.columns[j],
                        'correlation': corr_value,
                        'type': 'positive' if corr_value > 0 else 'negative'
                    })
        
        return strong_correlations
    
    def create_scatter_plots(self, metric_pairs=None):
        """Create scatter plots for correlated metrics"""
        if metric_pairs is None:
            metric_pairs = [('production_rate', 'quality_score'),
                          ('maintenance_score', 'downtime_hours'),
                          ('production_rate', 'energy_kwh')]
        
        figs = []
        
        for metric1, metric2 in metric_pairs:
            fig, ax = plt.subplots(figsize=(8, 6))
            
            ax.scatter(self.data[metric1], self.data[metric2], alpha=0.6, s=50)
            
            # Add regression line
            slope, intercept, r_value, p_value, std_err = stats.linregress(
                self.data[metric1], self.data[metric2]
            )
            
            x_line = np.array([self.data[metric1].min(), self.data[metric1].max()])
            y_line = slope * x_line + intercept
            
            ax.plot(x_line, y_line, color='red', linewidth=2,
                   label=f'y = {slope:.2f}x + {intercept:.2f}\nR² = {r_value**2:.2f}')
            
            ax.set_xlabel(metric1.replace('_', ' ').title())
            ax.set_ylabel(metric2.replace('_', ' ').title())
            ax.set_title(f'🔗 {metric1.replace("_", " ").title()} vs {metric2.replace("_", " ").title()}', 
                        fontsize=12, fontweight='bold')
            ax.legend()
            ax.grid(True, alpha=0.3)
            
            figs.append(fig)
        
        return figs

import asyncio
import time
from datetime import datetime, timedelta
from collections import defaultdict
from typing import Dict, List, Optional
import hashlib

class ManufacturingAISystem:
    """Advanced AI system for manufacturing with production patterns"""
    
    def __init__(self):
        self.request_log = []
        self.cost_tracker = defaultdict(float)
        self.rate_limiter = RateLimiter(max_calls=100, period=60)  # 100 calls/minute
        self.model_versions = {
            'quality_inspection': 'v2.1',
            'predictive_maintenance': 'v1.4',
            'production_optimization': 'v3.0'
        }
    
    async def async_process_production_batch(self, batch_data: List[Dict]):
        """Process production batch asynchronously (OpenAI-style pattern)"""
        print(f"🏭 Starting async processing of {len(batch_data)} items...")
        
        # Create tasks for parallel processing
        tasks = []
        for item in batch_data:
            task = self.process_manufacturing_item(item)
            tasks.append(task)
        
        # Process in parallel (like OpenAI's async API)
        start_time = time.time()
        results = await asyncio.gather(*tasks, return_exceptions=True)
        processing_time = time.time() - start_time
        
        # Aggregate results
        successful = [r for r in results if not isinstance(r, Exception)]
        failed = [r for r in results if isinstance(r, Exception)]
        
        return {
            'total_processed': len(batch_data),
            'successful': len(successful),
            'failed': len(failed),
            'processing_time': processing_time,
            'avg_time_per_item': processing_time / len(batch_data),
            'results': successful[:5]  # Return first 5 results for demo
        }
    
    async def process_manufacturing_item(self, item: Dict):
        """Process single manufacturing item with AI"""
        # Simulate AI processing delay
        await asyncio.sleep(0.1)
        
        # Apply rate limiting
        if not self.rate_limiter.allow_request():
            raise Exception("Rate limit exceeded")
        
        # Track cost (simulated)
        cost = self.track_ai_cost(item)
        
        # Process based on item type
        if item.get('type') == 'quality_inspection':
            result = self.run_quality_inspection(item)
        elif item.get('type') == 'predictive_maintenance':
            result = self.run_predictive_maintenance(item)
        else:
            result = self.run_general_analysis(item)
        
        # Log request
        self.log_request(item, result, cost)
        
        return {
            'item_id': item.get('id'),
            'result': result,
            'model_version': self.model_versions.get(item.get('type', 'general'), 'v1.0'),
            'cost': cost,
            'timestamp': datetime.now().isoformat()
        }
    
    def run_quality_inspection(self, item: Dict):
        """Run quality inspection AI model"""
        # Simulate AI inference
        defects = ['scratch', 'dent', 'misalignment']
        
        return {
            'defects_found': len([d for d in defects if d in str(item)]),
            'quality_score': max(0, 100 - len(defects) * 15),
            'pass': len([d for d in defects if d in str(item)]) == 0,
            'confidence': 0.92
        }
    
    def track_ai_cost(self, item: Dict) -> float:
        """Track AI inference cost (simplified)"""
        cost_rates = {
            'quality_inspection': 0.001,  # $ per inference
            'predictive_maintenance': 0.002,
            'production_optimization': 0.003
        }
        
        item_type = item.get('type', 'general')
        base_cost = cost_rates.get(item_type, 0.0005)
        
        # Add complexity factor
        complexity = len(json.dumps(item)) / 1000  # Size factor
        cost = base_cost * (1 + complexity)
        
        self.cost_tracker[item_type] += cost
        return cost
    
    def log_request(self, item: Dict, result: Dict, cost: float):
        """Log AI request for auditing and analytics"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'item_id': item.get('id'),
            'item_type': item.get('type'),
            'model_version': self.model_versions.get(item.get('type'), 'v1.0'),
            'cost': cost,
            'result_summary': str(result)[:100]  # First 100 chars
        }
        
        self.request_log.append(log_entry)
        
        # Keep only last 1000 entries
        if len(self.request_log) > 1000:
            self.request_log = self.request_log[-1000:]
    
    def get_system_metrics(self):
        """Get system performance metrics"""
        if not self.request_log:
            return {}
        
        # Calculate metrics
        total_cost = sum(self.cost_tracker.values())
        avg_cost = total_cost / len(self.request_log) if self.request_log else 0
        
        # Group by model version
        version_usage = defaultdict(int)
        for log in self.request_log:
            version_usage[log['model_version']] += 1
        
        return {
            'total_requests': len(self.request_log),
            'total_cost': total_cost,
            'avg_cost_per_request': avg_cost,
            'cost_by_type': dict(self.cost_tracker),
            'version_usage': dict(version_usage),
            'rate_limit_status': self.rate_limiter.get_status()
        }

class RateLimiter:
    """Rate limiter for AI API calls"""
    
    def __init__(self, max_calls: int = 100, period: int = 60):
        self.max_calls = max_calls
        self.period = period  # seconds
        self.calls = []
    
    def allow_request(self) -> bool:
        """Check if request is allowed under rate limit"""
        now = time.time()
        
        # Remove old calls
        self.calls = [call for call in self.calls if now - call < self.period]
        
        # Check if under limit
        if len(self.calls) < self.max_calls:
            self.calls.append(now)
            return True
        
        return False
    
    def get_status(self) -> Dict:
        """Get current rate limit status"""
        now = time.time()
        self.calls = [call for call in self.calls if now - call < self.period]
        
        return {
            'current_calls': len(self.calls),
            'max_calls': self.max_calls,
            'period_seconds': self.period,
            'remaining_calls': self.max_calls - len(self.calls),
            'reset_in_seconds': self.period - (now - min(self.calls)) if self.calls else 0
        }
